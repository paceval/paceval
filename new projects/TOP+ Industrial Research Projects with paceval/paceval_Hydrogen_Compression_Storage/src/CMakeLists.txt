cmake_minimum_required(VERSION 3.16)
project(paceval_hydrogen_thermal_envelope LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Set -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_runtime SHARED IMPORTED GLOBAL)
set_target_properties(paceval_runtime PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(hydrogen_thermal_kernels hydrogen_kernels.cpp paceval_main.cpp)
target_link_libraries(hydrogen_thermal_kernels PRIVATE paceval_runtime dl)
set_target_properties(hydrogen_thermal_kernels PROPERTIES BUILD_RPATH "$<TARGET_FILE_DIR:paceval_runtime>")
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(hydrogen_thermal_kernels PRIVATE -O2)
endif()
enable_testing()
add_test(NAME original_paceval_sdk_csv_and_operator_tests COMMAND hydrogen_thermal_kernels --selftest)
add_test(NAME original_sdk_compression_small COMMAND hydrogen_thermal_kernels --run 120 4 "${CMAKE_BINARY_DIR}/small_result")
add_test(NAME original_sdk_compression_medium COMMAND hydrogen_thermal_kernels --run 512 6 "${CMAKE_BINARY_DIR}/medium_result")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME independent_high_precision_reference COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../validation/check_high_precision.py" "${CMAKE_CURRENT_SOURCE_DIR}/../results")
add_test(NAME all_archive_markdown_math_and_links COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_archive_markdown.py")
add_test(NAME publication_and_research_integrity COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_publication.py")
add_test(NAME result_integrity_metadata COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_evidence.py")
add_test(NAME sha256_file_manifest COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_manifest.py")
