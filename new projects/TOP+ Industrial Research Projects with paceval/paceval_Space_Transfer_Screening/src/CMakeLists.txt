cmake_minimum_required(VERSION 3.16)
project(paceval_space_transfer_screening LANGUAGES CXX)
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Specify -DPACEVAL_RUNTIME=/absolute/path/to/compatible/libpaceval_linux_sharedLIB.so")
endif()
add_executable(space_transfer space_transfer.cpp paceval_main.cpp)
target_compile_features(space_transfer PRIVATE cxx_std_17)
target_include_directories(space_transfer PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(space_transfer PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(space_transfer PRIVATE "${PACEVAL_RUNTIME}")
if(UNIX AND NOT APPLE)
 get_filename_component(PACEVAL_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
 set_target_properties(space_transfer PROPERTIES BUILD_RPATH "${PACEVAL_DIR}")
endif()
enable_testing()
add_test(NAME space_transfer_actual_paceval COMMAND space_transfer --cases 180 --out "${CMAKE_CURRENT_BINARY_DIR}/test_results")
