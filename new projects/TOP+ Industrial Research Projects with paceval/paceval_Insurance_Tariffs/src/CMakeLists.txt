cmake_minimum_required(VERSION 3.16)
project(paceval_insurance_tariff_research LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
 message(FATAL_ERROR "Use the compatible architecture-specific paceval runtime; this release is Linux-validated")
endif()
if(NOT PACEVAL_RUNTIME AND DEFINED ENV{PACEVAL_LIB})
 set(PACEVAL_RUNTIME "$ENV{PACEVAL_LIB}")
endif()
if(NOT PACEVAL_RUNTIME OR NOT IS_ABSOLUTE "${PACEVAL_RUNTIME}" OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Pass -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_runtime SHARED IMPORTED GLOBAL)
set_target_properties(paceval_runtime PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
# paceval_main.cpp is the official SDK API facade, not the actual arithmetic runtime.
add_executable(insurance_tariffs insurance_tariffs.cpp paceval_main.cpp)
target_include_directories(insurance_tariffs PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(insurance_tariffs PRIVATE paceval_runtime)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(insurance_tariffs PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
 add_test(NAME native_input_validation COMMAND insurance_tariffs --self-test)
 add_test(NAME sdk_100_models COMMAND insurance_tariffs --models 100 --pairs 24 --batch 37 --out "${CMAKE_CURRENT_BINARY_DIR}/test100")
 add_test(NAME sdk_small_batch COMMAND insurance_tariffs --models 24 --pairs 24 --batch 1 --out "${CMAKE_CURRENT_BINARY_DIR}/test_batch1")
 add_test(NAME sdk_csv_roundtrip COMMAND insurance_tariffs --models 4 --input "${CMAKE_CURRENT_SOURCE_DIR}/../examples/policies.csv" --batch 32 --out "${CMAKE_CURRENT_BINARY_DIR}/test_import")
 add_test(NAME reject_invalid_csv COMMAND insurance_tariffs --models 4 --input "${CMAKE_CURRENT_SOURCE_DIR}/../examples/invalid_policy.csv" --out "${CMAKE_CURRENT_BINARY_DIR}/bad")
 set_tests_properties(reject_invalid_csv PROPERTIES WILL_FAIL TRUE)
endif()
