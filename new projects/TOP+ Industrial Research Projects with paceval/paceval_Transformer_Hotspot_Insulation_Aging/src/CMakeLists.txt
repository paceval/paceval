cmake_minimum_required(VERSION 3.16)
project(paceval_transformer_hotspot LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Please specify -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_external SHARED IMPORTED)
set_target_properties(paceval_external PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(transformer_hotspot transformer_main.cpp paceval_main.cpp)
target_include_directories(transformer_hotspot PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(transformer_hotspot PRIVATE paceval_external)
if(UNIX AND NOT APPLE)
 target_link_libraries(transformer_hotspot PRIVATE dl pthread)
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME sdk_scalar_batch_operators_conversions COMMAND transformer_hotspot --selftest)
add_test(NAME full_csv_to_paceval_to_csv COMMAND transformer_hotspot --run "${CMAKE_CURRENT_LIST_DIR}/../input/transformer_profiles.csv" "${CMAKE_CURRENT_LIST_DIR}/../input/transformer_parameters.csv" "${CMAKE_CURRENT_BINARY_DIR}/test_results")
add_test(NAME high_precision_thermal_model COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/verify_high_precision.py" "${CMAKE_CURRENT_BINARY_DIR}/test_results" "${CMAKE_CURRENT_LIST_DIR}/../input")
set_tests_properties(high_precision_thermal_model PROPERTIES DEPENDS full_csv_to_paceval_to_csv)
add_test(NAME readme_link_and_syntax_checker COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/check_readme_math.py" "${CMAKE_CURRENT_LIST_DIR}/../README.md")
add_test(NAME readme_checker_negative_tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/check_readme_math.py" --selftest)
