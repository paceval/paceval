cmake_minimum_required(VERSION 3.16)
project(paceval_battery_charging_aging LANGUAGES CXX)
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
    message(FATAL_ERROR "Set -DPACEVAL_RUNTIME=/absolute/path/to/compatible/libpaceval_linux_sharedLIB.so")
endif()
add_executable(battery_charging_aging battery_charging_aging.cpp paceval_main.cpp)
target_compile_features(battery_charging_aging PRIVATE cxx_std_17)
target_include_directories(battery_charging_aging PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(battery_charging_aging PRIVATE "${PACEVAL_RUNTIME}")
if(UNIX AND NOT APPLE)
    get_filename_component(PACEVAL_LIB_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
    set_target_properties(battery_charging_aging PROPERTIES BUILD_RPATH "${PACEVAL_LIB_DIR}")
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(battery_charging_aging PRIVATE -O2 -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME actual_paceval_sdk_battery_schedule COMMAND battery_charging_aging --out "${CMAKE_CURRENT_BINARY_DIR}/ctest_run" --profiles 1 --weights 1 --surface 9)
