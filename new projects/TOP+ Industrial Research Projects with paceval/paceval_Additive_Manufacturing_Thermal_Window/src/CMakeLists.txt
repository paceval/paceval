cmake_minimum_required(VERSION 3.18)
project(paceval_additive_thermal_window LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PACEVAL_RUNTIME "" CACHE FILEPATH "Original ABI compatible Linux x86-64 paceval shared library")
if(NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Missing ORIGINAL compatible paceval .so, supply -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(am_thermal_paceval am_thermal_paceval.cpp paceval_main.cpp)
target_link_libraries(am_thermal_paceval PRIVATE "${PACEVAL_RUNTIME}" dl pthread)
target_compile_options(am_thermal_paceval PRIVATE -Wall -Wextra -Wpedantic -O2)
get_filename_component(PACEVAL_LIB_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
set_target_properties(am_thermal_paceval PROPERTIES BUILD_RPATH "${PACEVAL_LIB_DIR}")
enable_testing()
add_test(NAME paceval_sdk_operator_conversion_selftest COMMAND am_thermal_paceval "${CMAKE_SOURCE_DIR}/.." --selftest)
add_test(NAME paceval_sdk_end_to_end_csv_batch_tinc COMMAND am_thermal_paceval "${CMAKE_SOURCE_DIR}/..")
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME independent_mpmath_audit COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/../validation/high_precision_reference.py" "${CMAKE_SOURCE_DIR}/..")
  add_test(NAME readme_and_artifact_integrity COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/../validation/check_project.py" "${CMAKE_SOURCE_DIR}/..")
endif()
