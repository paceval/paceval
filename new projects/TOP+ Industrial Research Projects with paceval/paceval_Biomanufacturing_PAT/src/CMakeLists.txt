cmake_minimum_required(VERSION 3.18)
project(paceval_biomanufacturing_pat LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PACEVAL_RUNTIME "" CACHE FILEPATH "Original ABI-compatible paceval Linux shared library")
if(NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Original paceval shared library required: -DPACEVAL_RUNTIME=/absolute/path/libpaceval_linux_sharedLIB.so")
endif()
add_executable(biomanufacturing_pat biomanufacturing_pat.cpp paceval_main.cpp)
target_compile_options(biomanufacturing_pat PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(biomanufacturing_pat PRIVATE "${PACEVAL_RUNTIME}" dl pthread)
get_filename_component(RUNTIME_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
set_target_properties(biomanufacturing_pat PROPERTIES BUILD_RPATH "${RUNTIME_DIR}")
enable_testing()
add_test(NAME sdk_csv_conversion_operator_selftest COMMAND biomanufacturing_pat "${CMAKE_SOURCE_DIR}/.." --selftest)
add_test(NAME sdk_end_to_end_graphs_batch_tinc_csv COMMAND biomanufacturing_pat "${CMAKE_SOURCE_DIR}/..")
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
 add_test(NAME high_precision_reference COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/../validation/high_precision_reference.py" "${CMAKE_SOURCE_DIR}/..")
 add_test(NAME project_integrity COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/../validation/check_project.py" "${CMAKE_SOURCE_DIR}/..")
endif()
