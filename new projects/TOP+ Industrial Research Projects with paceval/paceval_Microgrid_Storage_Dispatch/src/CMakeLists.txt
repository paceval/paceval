cmake_minimum_required(VERSION 3.16)
project(paceval_microgrid_dispatch LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Please specify -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_external SHARED IMPORTED)
set_target_properties(paceval_external PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(microgrid_dispatch microgrid_main.cpp paceval_main.cpp)
target_include_directories(microgrid_dispatch PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(microgrid_dispatch PRIVATE paceval_external)
if(UNIX AND NOT APPLE)
 target_link_libraries(microgrid_dispatch PRIVATE dl pthread)
endif()
enable_testing()
add_test(NAME sdk_scalar_batch_and_operators COMMAND microgrid_dispatch --selftest)
add_test(NAME full_csv_to_paceval_to_csv_pipeline COMMAND microgrid_dispatch --run "${CMAKE_CURRENT_LIST_DIR}/../input/dispatch_candidates.csv" "${CMAKE_CURRENT_LIST_DIR}/../input/parameters.csv" "${CMAKE_CURRENT_BINARY_DIR}/integration_results")
add_test(NAME independent_high_precision_verification COMMAND python3 "${CMAKE_CURRENT_LIST_DIR}/../scripts/verify_high_precision.py" "${CMAKE_CURRENT_BINARY_DIR}/integration_results" "${CMAKE_CURRENT_LIST_DIR}/../input")
set_tests_properties(independent_high_precision_verification PROPERTIES DEPENDS full_csv_to_paceval_to_csv_pipeline)
add_test(NAME readme_links_math_checker COMMAND python3 "${CMAKE_CURRENT_LIST_DIR}/check_readme_math.py" "${CMAKE_CURRENT_LIST_DIR}/../README.md")
add_test(NAME readme_checker_negative_tests COMMAND python3 "${CMAKE_CURRENT_LIST_DIR}/check_readme_math.py" --selftest)
