cmake_minimum_required(VERSION 3.16)
project(paceval_fermi_2fle_research LANGUAGES CXX)
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Specify -DPACEVAL_RUNTIME=/absolute/path/to/compatible/libpaceval_linux_sharedLIB.so")
endif()
add_executable(fermi_2fle_demo fermi_2fle_demo.cpp paceval_main.cpp)
target_compile_features(fermi_2fle_demo PRIVATE cxx_std_17)
target_include_directories(fermi_2fle_demo PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(fermi_2fle_demo PRIVATE -Wall -Wextra -Wpedantic -O2)
target_link_libraries(fermi_2fle_demo PRIVATE "${PACEVAL_RUNTIME}")
if(UNIX AND NOT APPLE)
 get_filename_component(PACEVAL_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
 set_target_properties(fermi_2fle_demo PROPERTIES BUILD_RPATH "${PACEVAL_DIR}")
endif()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/test_out/examples")
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/../examples/unidentified_60_from_user_whitepaper.csv" "${CMAKE_CURRENT_BINARY_DIR}/test_out/examples/unidentified_60_from_user_whitepaper.csv" COPYONLY)
enable_testing()
add_test(NAME fermi_2fle_real_sdk COMMAND fermi_2fle_demo --out "${CMAKE_CURRENT_BINARY_DIR}/test_out" --sources 6 --bins 10 --grid 13)
