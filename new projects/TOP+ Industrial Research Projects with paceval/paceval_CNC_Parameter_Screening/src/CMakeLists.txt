cmake_minimum_required(VERSION 3.16)
project(paceval_cnc_screening LANGUAGES CXX)
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Specify -DPACEVAL_RUNTIME=/absolute/path/to/compatible/libpaceval_linux_sharedLIB.so")
endif()
add_executable(cnc_screening cnc_screening.cpp paceval_main.cpp)
target_compile_features(cnc_screening PRIVATE cxx_std_17)
target_compile_options(cnc_screening PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(cnc_screening PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(cnc_screening PRIVATE "${PACEVAL_RUNTIME}")
get_filename_component(PACEVAL_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
set_target_properties(cnc_screening PROPERTIES BUILD_RPATH "${PACEVAL_DIR}")
enable_testing()
add_test(NAME cnc_sdk_integration COMMAND cnc_screening --out "${CMAKE_CURRENT_BINARY_DIR}/test_results")
