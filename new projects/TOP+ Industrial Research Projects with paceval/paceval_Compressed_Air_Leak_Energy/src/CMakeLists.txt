cmake_minimum_required(VERSION 3.16)
project(paceval_compressed_air LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Provide -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_external SHARED IMPORTED)
set_target_properties(paceval_external PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(compressed_air compressed_air_main.cpp paceval_main.cpp)
target_include_directories(compressed_air PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(compressed_air PRIVATE paceval_external)
if(UNIX AND NOT APPLE)
 target_link_libraries(compressed_air PRIVATE dl pthread)
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME sdk_scalar_batch_operators_conversion COMMAND compressed_air --selftest)
add_test(NAME full_csv_paceval_pipeline COMMAND compressed_air --run "${CMAKE_CURRENT_LIST_DIR}/../input/cases.csv" "${CMAKE_CURRENT_BINARY_DIR}/ctest_results")
add_test(NAME high_precision_independent_reference COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/verify_high_precision.py" "${CMAKE_CURRENT_BINARY_DIR}/ctest_results" "${CMAKE_CURRENT_LIST_DIR}/../input/cases.csv")
set_tests_properties(high_precision_independent_reference PROPERTIES DEPENDS full_csv_paceval_pipeline)
add_test(NAME archive_markdown_math_links COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/check_markdown.py" "${CMAKE_CURRENT_LIST_DIR}/..")
add_test(NAME markdown_negative_regression COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/check_markdown.py" --selftest)
