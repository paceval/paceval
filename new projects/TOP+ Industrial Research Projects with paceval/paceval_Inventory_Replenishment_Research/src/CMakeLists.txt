cmake_minimum_required(VERSION 3.16)
project(paceval_inventory_replenishment LANGUAGES CXX)
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Specify -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(inventory_replenishment inventory_replenishment.cpp paceval_main.cpp)
target_compile_features(inventory_replenishment PRIVATE cxx_std_17)
target_include_directories(inventory_replenishment PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(inventory_replenishment PRIVATE "${PACEVAL_RUNTIME}")
if(MSVC)
 target_compile_options(inventory_replenishment PRIVATE /W4)
else()
 target_compile_options(inventory_replenishment PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(UNIX AND NOT APPLE)
 get_filename_component(RUNTIME_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
 set_target_properties(inventory_replenishment PROPERTIES BUILD_RPATH "${RUNTIME_DIR}")
endif()
enable_testing()
add_test(NAME sdk_inventory_integration COMMAND inventory_replenishment --test --out "${CMAKE_CURRENT_BINARY_DIR}/test_outputs")
