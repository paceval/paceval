cmake_minimum_required(VERSION 3.16)
project(paceval_ecg_feature_rules LANGUAGES CXX)
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Supply -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(ecg_rules ecg_rules.cpp paceval_main.cpp)
target_compile_features(ecg_rules PRIVATE cxx_std_17)
target_include_directories(ecg_rules PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(ecg_rules PRIVATE "${PACEVAL_RUNTIME}")
target_compile_options(ecg_rules PRIVATE -Wall -Wextra -Wpedantic)
if(UNIX AND NOT APPLE)
 get_filename_component(RUNTIME_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
 set_target_properties(ecg_rules PROPERTIES BUILD_RPATH "${RUNTIME_DIR}")
endif()
enable_testing()
add_test(NAME ecg_sdk_integration COMMAND ecg_rules --records 300 --out "${CMAKE_CURRENT_BINARY_DIR}/test_results")
