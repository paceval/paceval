cmake_minimum_required(VERSION 3.18)
project(paceval_district_heating_thermal_dispatch LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PACEVAL_RUNTIME "" CACHE FILEPATH "Original ABI-compatible paceval Linux shared library")
if(NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Original shared runtime required: -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(district_heating_paceval district_heating_paceval.cpp paceval_main.cpp)
target_compile_options(district_heating_paceval PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(district_heating_paceval PRIVATE "${PACEVAL_RUNTIME}" dl pthread)
get_filename_component(PACEVAL_RUNTIME_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
set_target_properties(district_heating_paceval PROPERTIES BUILD_RPATH "${PACEVAL_RUNTIME_DIR}")
add_executable(district_conversion_test test_numeric_conversions.cpp paceval_main.cpp)
add_executable(district_operator_test test_operator_semantics.cpp paceval_main.cpp)
foreach(TEST_EXE district_conversion_test district_operator_test)
 target_link_libraries(${TEST_EXE} PRIVATE "${PACEVAL_RUNTIME}" dl pthread)
 set_target_properties(${TEST_EXE} PROPERTIES BUILD_RPATH "${PACEVAL_RUNTIME_DIR}")
endforeach()
enable_testing()
add_test(NAME original_sdk_converter_unit COMMAND district_conversion_test)
add_test(NAME original_sdk_operators_unit COMMAND district_operator_test)
add_test(NAME sdk_conversion_and_operators COMMAND district_heating_paceval "${CMAKE_SOURCE_DIR}/.." --selftest)
add_test(NAME sdk_complete_network_edge_batch COMMAND district_heating_paceval "${CMAKE_SOURCE_DIR}/..")
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
 add_test(NAME independent_high_precision_audit COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/../validation/high_precision_reference.py" "${CMAKE_SOURCE_DIR}/..")
 add_test(NAME research_artifacts_checker COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/../validation/check_project.py" "${CMAKE_SOURCE_DIR}/..")
endif()
