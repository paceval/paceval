cmake_minimum_required(VERSION 3.16)
project(paceval_explainable_anomaly_detection LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The validated demonstrator targets Linux with the original paceval runtime")
endif()
if(NOT PACEVAL_RUNTIME AND DEFINED ENV{PACEVAL_LIB})
  set(PACEVAL_RUNTIME "$ENV{PACEVAL_LIB}")
endif()
if(NOT PACEVAL_RUNTIME OR NOT IS_ABSOLUTE "${PACEVAL_RUNTIME}" OR NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Set -DPACEVAL_RUNTIME=/absolute/path/to/original_paceval.so or PACEVAL_LIB")
endif()
add_library(paceval_original SHARED IMPORTED GLOBAL)
set_target_properties(paceval_original PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(anomaly_detection anomaly_detection.cpp paceval_main.cpp)
target_include_directories(anomaly_detection PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(anomaly_detection PRIVATE paceval_original)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(anomaly_detection PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME sdk_and_100_graphs COMMAND anomaly_detection --self-test)
add_test(NAME full_synthetic_replay COMMAND anomaly_detection --out "${CMAKE_CURRENT_BINARY_DIR}/generated" --batch 37)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
 set_tests_properties(full_synthetic_replay PROPERTIES FIXTURES_SETUP generated_files)
 add_test(NAME independent_high_precision_and_metrics COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/validate_outputs.py" --examples "${CMAKE_CURRENT_BINARY_DIR}/generated")
 set_tests_properties(independent_high_precision_and_metrics PROPERTIES FIXTURES_REQUIRED generated_files)
 add_test(NAME strict_csv_replay_and_rejection COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_csv.py" --binary "$<TARGET_FILE:anomaly_detection>" --examples "${CMAKE_CURRENT_SOURCE_DIR}/../examples")
endif()
message(STATUS "Using external original paceval runtime: ${PACEVAL_RUNTIME}")
