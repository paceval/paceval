cmake_minimum_required(VERSION 3.16)
project(paceval_agent_financial_arithmetic LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Linux compatible paceval runtime required")
endif()
if(NOT PACEVAL_RUNTIME AND DEFINED ENV{PACEVAL_LIB})
  set(PACEVAL_RUNTIME "$ENV{PACEVAL_LIB}")
endif()
if(NOT PACEVAL_RUNTIME OR NOT IS_ABSOLUTE "${PACEVAL_RUNTIME}" OR NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Set -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_runtime SHARED IMPORTED GLOBAL)
set_target_properties(paceval_runtime PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(financial_agent financial_agent.cpp paceval_main.cpp)
target_include_directories(financial_agent PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(financial_agent PRIVATE paceval_runtime)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(financial_agent PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME native_financial_math COMMAND financial_agent --self-test)
  add_test(NAME sdk_financial_cases COMMAND financial_agent --demo --batch 37 --out "${CMAKE_CURRENT_BINARY_DIR}/suite")
  add_test(NAME sdk_batch_one COMMAND financial_agent --demo --batch 1 --out "${CMAKE_CURRENT_BINARY_DIR}/batch_one")
  # Automated 100-digit independent audit including EXACT binary64 TINC input
  # interpretation. A missed interval endpoint makes this CTest FAIL.
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME independent_high_precision_audit
           COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/high_precision_reference.py"
                   "${CMAKE_CURRENT_BINARY_DIR}/suite")
  set_tests_properties(sdk_financial_cases PROPERTIES FIXTURES_SETUP financial_demo_output)
  set_tests_properties(independent_high_precision_audit PROPERTIES FIXTURES_REQUIRED financial_demo_output)
endif()
