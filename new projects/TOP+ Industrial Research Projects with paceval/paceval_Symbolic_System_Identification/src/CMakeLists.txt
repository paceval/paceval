cmake_minimum_required(VERSION 3.16)
project(paceval_symbolic_system_identification LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This reference release has been tested on Linux")
endif()
if(NOT PACEVAL_RUNTIME AND DEFINED ENV{PACEVAL_LIB})
    set(PACEVAL_RUNTIME "$ENV{PACEVAL_LIB}")
endif()
if(NOT PACEVAL_RUNTIME OR NOT IS_ABSOLUTE "${PACEVAL_RUNTIME}" OR NOT EXISTS "${PACEVAL_RUNTIME}")
    message(FATAL_ERROR "Pass -DPACEVAL_RUNTIME=/absolute/path/libpaceval_linux_sharedLIB.so (or set PACEVAL_LIB)")
endif()
# The official paceval_main.cpp is the C/C++ interface forwarding layer.
# The compatible proprietary mathematical runtime must be supplied externally.
add_library(paceval_runtime SHARED IMPORTED GLOBAL)
set_target_properties(paceval_runtime PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(system_identification system_identification.cpp paceval_main.cpp)
target_include_directories(system_identification PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(system_identification PRIVATE paceval_runtime)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(system_identification PRIVATE -Wall -Wextra -Wpedantic)
    if(CMAKE_BUILD_TYPE STREQUAL "Release")
        target_compile_options(system_identification PRIVATE -O3)
    endif()
endif()
include(CTest)
if(BUILD_TESTING)
    add_test(NAME native_oracle_self_test COMMAND system_identification --self-test)
    add_test(NAME real_paceval_small COMMAND system_identification --candidates 128 --batch 37 --output "${CMAKE_CURRENT_BINARY_DIR}/smoke")
    add_test(NAME real_paceval_medium COMMAND system_identification --candidates 256 --batch 128 --output "${CMAKE_CURRENT_BINARY_DIR}/extended")
endif()
