cmake_minimum_required(VERSION 3.16)
project(paceval_heat_pump_twin LANGUAGES CXX)
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Specify -DPACEVAL_RUNTIME=/absolute/path/to/compatible/libpaceval_linux_sharedLIB.so")
endif()
add_executable(heat_pump_twin heat_pump_twin.cpp paceval_main.cpp)
target_compile_features(heat_pump_twin PRIVATE cxx_std_17)
target_include_directories(heat_pump_twin PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(heat_pump_twin PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(heat_pump_twin PRIVATE "${PACEVAL_RUNTIME}")
if(UNIX AND NOT APPLE)
 get_filename_component(PACEVAL_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
 set_target_properties(heat_pump_twin PROPERTIES BUILD_RPATH "${PACEVAL_DIR}")
endif()
enable_testing()
add_test(NAME heat_pump_twin_real_paceval COMMAND heat_pump_twin --cases 180 --out "${CMAKE_CURRENT_BINARY_DIR}/test_results")
