cmake_minimum_required(VERSION 3.16)
project(paceval_building_thermal_twin LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PACEVAL_RUNTIME "" CACHE FILEPATH "External original paceval Linux x64 shared library (not redistributed)")
if(NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "An original Linux paceval library is required: set -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(building_thermal_twin thermal_twin.cpp paceval_main.cpp)
target_compile_options(building_thermal_twin PRIVATE -O2 -Wall -Wextra -Wpedantic)
target_link_libraries(building_thermal_twin PRIVATE "${PACEVAL_RUNTIME}" dl pthread)
get_filename_component(PACEVAL_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
set_target_properties(building_thermal_twin PROPERTIES BUILD_RPATH "${PACEVAL_DIR}")
enable_testing()
add_test(NAME building_thermal_twin_original_paceval COMMAND building_thermal_twin --out "${CMAKE_BINARY_DIR}/ctest" --steps 32 --grid 3)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set_tests_properties(building_thermal_twin_original_paceval PROPERTIES FIXTURES_SETUP generated_paceval_output)
add_test(NAME building_thermal_twin_80_digit_validation
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_results.py" --root "${CMAKE_BINARY_DIR}/ctest")
set_tests_properties(building_thermal_twin_80_digit_validation PROPERTIES FIXTURES_REQUIRED generated_paceval_output)
add_test(NAME building_thermal_twin_readme_math
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_readme_math.py")
