cmake_minimum_required(VERSION 3.16)
project(paceval_wp27_kinetic_discovery LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Set -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_runtime SHARED IMPORTED GLOBAL)
set_target_properties(paceval_runtime PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(kinetic_discovery kinetic_discovery.cpp paceval_main.cpp)
target_link_libraries(kinetic_discovery PRIVATE paceval_runtime dl)
set_target_properties(kinetic_discovery PROPERTIES BUILD_RPATH "$<TARGET_FILE_DIR:paceval_runtime>")
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(kinetic_discovery PRIVATE -O2 -Wall -Wextra)
endif()
enable_testing()
add_test(NAME sdk_conversions_and_operator_semantics COMMAND kinetic_discovery --selftest)
add_test(NAME sdk_search_small COMMAND kinetic_discovery --run 32 16 16 "${CMAKE_BINARY_DIR}/small_results")
add_test(NAME sdk_search_medium COMMAND kinetic_discovery --run 64 32 32 "${CMAKE_BINARY_DIR}/medium_results")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME independent_high_precision_validation COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../validation/check_high_precision.py" "${CMAKE_CURRENT_SOURCE_DIR}/../results")
add_test(NAME archive_github_markdown_math COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_archive_markdown.py")
add_test(NAME publication_integrity COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_publication.py")
add_test(NAME execution_evidence_research_contract COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_evidence.py")
add_test(NAME sha256_manifest COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_manifest.py")
