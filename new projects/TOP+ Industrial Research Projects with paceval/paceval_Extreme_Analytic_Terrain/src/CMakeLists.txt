cmake_minimum_required(VERSION 3.16)
project(paceval_extreme_analytic_terrain LANGUAGES CXX)
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Specify -DPACEVAL_RUNTIME=/absolute/path/to/compatible/libpaceval_linux_sharedLIB.so")
endif()
add_executable(extreme_terrain extreme_terrain.cpp paceval_main.cpp)
target_compile_features(extreme_terrain PRIVATE cxx_std_17)
target_include_directories(extreme_terrain PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(extreme_terrain PRIVATE "${PACEVAL_RUNTIME}")
target_compile_options(extreme_terrain PRIVATE -O2 -Wall -Wextra -Wpedantic)
if(UNIX AND NOT APPLE)
 get_filename_component(PACEVAL_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
 set_target_properties(extreme_terrain PROPERTIES BUILD_RPATH "${PACEVAL_DIR}")
endif()
enable_testing()
add_test(NAME terrain_real_sdk COMMAND extreme_terrain --out "${CMAKE_CURRENT_BINARY_DIR}/test_terrain" --quick)
