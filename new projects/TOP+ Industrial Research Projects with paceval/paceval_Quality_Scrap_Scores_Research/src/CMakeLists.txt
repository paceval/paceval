cmake_minimum_required(VERSION 3.16)
project(paceval_quality_scrap_scores LANGUAGES CXX)
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Specify -DPACEVAL_RUNTIME=/absolute/path/to/compatible/libpaceval.so")
endif()
add_executable(quality_scores quality_scores.cpp paceval_main.cpp)
target_compile_features(quality_scores PRIVATE cxx_std_17)
target_include_directories(quality_scores PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(quality_scores PRIVATE "${PACEVAL_RUNTIME}")
if(NOT MSVC)
 target_compile_options(quality_scores PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(UNIX AND NOT APPLE)
 get_filename_component(RUNTIME_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
 set_target_properties(quality_scores PROPERTIES BUILD_RPATH "${RUNTIME_DIR}")
endif()
enable_testing()
add_test(NAME sdk_quality_integration COMMAND quality_scores --out "${CMAKE_CURRENT_BINARY_DIR}/test_outputs")
