cmake_minimum_required(VERSION 3.16)
project(paceval_hydraulic_pump_twin LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PACEVAL_RUNTIME "" CACHE FILEPATH "Original paceval Linux shared library, externally supplied")
if(NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Supply -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(hydraulic_pump_twin hydraulic_pump_twin.cpp paceval_main.cpp)
target_compile_options(hydraulic_pump_twin PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(hydraulic_pump_twin PRIVATE "${PACEVAL_RUNTIME}" dl pthread)
get_filename_component(PACEVAL_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
set_target_properties(hydraulic_pump_twin PROPERTIES BUILD_RPATH "${PACEVAL_DIR}")
enable_testing()
add_test(NAME original_paceval_sdk_integration COMMAND hydraulic_pump_twin --out "${CMAKE_CURRENT_BINARY_DIR}/ctest" --surface 9 --sweep 60 --hours 24)
