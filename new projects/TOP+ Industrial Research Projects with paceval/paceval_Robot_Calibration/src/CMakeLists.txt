cmake_minimum_required(VERSION 3.16)
project(paceval_robot_calibration LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The validated demonstrator targets the original Linux paceval runtime")
endif()
if(NOT PACEVAL_RUNTIME AND DEFINED ENV{PACEVAL_LIB})
  set(PACEVAL_RUNTIME "$ENV{PACEVAL_LIB}")
endif()
if(NOT PACEVAL_RUNTIME OR NOT IS_ABSOLUTE "${PACEVAL_RUNTIME}" OR NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Set -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so or PACEVAL_LIB")
endif()
add_library(paceval_original SHARED IMPORTED GLOBAL)
set_target_properties(paceval_original PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(robot_calibration robot_calibration.cpp paceval_main.cpp)
target_include_directories(robot_calibration PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(robot_calibration PRIVATE paceval_original)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(robot_calibration PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME independent_forward_kinematics COMMAND robot_calibration --self-test)
add_test(NAME offline_calibration_and_observability COMMAND robot_calibration --out "${CMAKE_CURRENT_BINARY_DIR}/test_output" --batch 37)
message(STATUS "Using separately provided original paceval runtime: ${PACEVAL_RUNTIME}")
# Python reference tests require mpmath and numpy (see README). The executable
# and first two tests remain buildable without any Python packages.
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  set_tests_properties(offline_calibration_and_observability PROPERTIES FIXTURES_SETUP generated_samples)
  add_test(NAME independent_100_digit_audit COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/validate_outputs.py" --examples "${CMAKE_CURRENT_BINARY_DIR}/test_output")
  set_tests_properties(independent_100_digit_audit PROPERTIES FIXTURES_REQUIRED generated_samples)
  add_test(NAME strict_csv_replay_and_rejection COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/test_csv_validation.py"
    --binary "$<TARGET_FILE:robot_calibration>"
    --examples "${CMAKE_CURRENT_SOURCE_DIR}/../examples")
endif()
