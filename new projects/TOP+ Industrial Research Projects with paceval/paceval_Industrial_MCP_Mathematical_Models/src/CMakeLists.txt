cmake_minimum_required(VERSION 3.18)
project(industrial_mcp_mathematical_models LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PACEVAL_RUNTIME "" CACHE FILEPATH "Path to original Linux paceval shared library")
if(NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Provide -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(industrial_mcp_paceval industrial_mcp_paceval.cpp paceval_main.cpp)
target_compile_options(industrial_mcp_paceval PRIVATE -O2 -Wall -Wextra -pedantic)
target_link_libraries(industrial_mcp_paceval PRIVATE "${PACEVAL_RUNTIME}" dl pthread)
get_filename_component(PACEVAL_LIB_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
set_target_properties(industrial_mcp_paceval PROPERTIES BUILD_RPATH "${PACEVAL_LIB_DIR}")
enable_testing()
add_test(NAME original_paceval_sdk_full_pipeline COMMAND industrial_mcp_paceval ${CMAKE_SOURCE_DIR}/.. --verify-only)

find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
 add_test(NAME rejects_invalid_measurement_inputs COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/test_invalid_inputs.py ${CMAKE_SOURCE_DIR}/.. $<TARGET_FILE:industrial_mcp_paceval>)
 add_test(NAME high_precision_scientific_audit COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/independent_reference.py ${CMAKE_SOURCE_DIR}/..)
 add_test(NAME live_mcp_record_validation COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/verify_live_mcp.py ${CMAKE_SOURCE_DIR}/..)
 add_test(NAME project_readme_math_and_links COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/check_readme_math.py ${CMAKE_SOURCE_DIR}/..)
 add_test(NAME model_formula_sha256_manifest COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/verify_manifest.py ${CMAKE_SOURCE_DIR}/..)
endif()
