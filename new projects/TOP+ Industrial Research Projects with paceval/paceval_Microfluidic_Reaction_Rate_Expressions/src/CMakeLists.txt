cmake_minimum_required(VERSION 3.16)
project(paceval_microfluidic_reaction_transport LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PACEVAL_RUNTIME "" CACHE FILEPATH "External original paceval Linux shared library (not redistributed)")
if(NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Provide an ORIGINAL paceval runtime using -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(microfluidic_reactor microfluidic_reactor.cpp paceval_main.cpp)
target_compile_options(microfluidic_reactor PRIVATE -O2 -Wall -Wextra -Wpedantic)
target_link_libraries(microfluidic_reactor PRIVATE "${PACEVAL_RUNTIME}" dl pthread)
get_filename_component(PACEVAL_LIB_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
set_target_properties(microfluidic_reactor PROPERTIES BUILD_RPATH "${PACEVAL_LIB_DIR}")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
enable_testing()
add_test(NAME original_paceval_microreactor COMMAND microfluidic_reactor --out "${CMAKE_BINARY_DIR}/test_run" --cells 16 --steps 120)
set_tests_properties(original_paceval_microreactor PROPERTIES FIXTURES_SETUP test_data)
add_test(NAME independent_numerical_validation COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_results.py" --root "${CMAKE_BINARY_DIR}/test_run")
set_tests_properties(independent_numerical_validation PROPERTIES FIXTURES_REQUIRED test_data)
add_test(NAME github_readme_math_and_links COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_readme_math.py")
