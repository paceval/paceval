cmake_minimum_required(VERSION 3.16)
project(paceval_parameterized_bulk_wave LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PACEVAL_RUNTIME "" CACHE FILEPATH "External original paceval Linux shared runtime: NOT redistributed")
if(NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Specify -DPACEVAL_RUNTIME=/absolute/path/libpaceval_linux_sharedLIB.so")
endif()
add_executable(bulk_wave_kernels bulk_wave_kernels.cpp paceval_main.cpp)
target_compile_options(bulk_wave_kernels PRIVATE -O2 -Wall -Wextra -Wpedantic)
target_link_libraries(bulk_wave_kernels PRIVATE "${PACEVAL_RUNTIME}" dl pthread)
get_filename_component(PACEVAL_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
set_target_properties(bulk_wave_kernels PROPERTIES BUILD_RPATH "${PACEVAL_DIR}")
enable_testing()
add_test(NAME original_paceval_bulk_wave_sdk COMMAND bulk_wave_kernels --out "${CMAKE_CURRENT_BINARY_DIR}/ctest" --n0 17 --time 0.08 --courant 0.32)
