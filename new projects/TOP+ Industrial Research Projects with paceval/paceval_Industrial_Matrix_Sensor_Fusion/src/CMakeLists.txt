cmake_minimum_required(VERSION 3.18)
project(paceval_industrial_matrix_sensor_fusion LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PACEVAL_RUNTIME "" CACHE FILEPATH "Original paceval Linux runtime shared library")
if(NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Provide -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(matrix_sensor_fusion matrix_sensor_fusion.cpp paceval_main.cpp)
target_compile_options(matrix_sensor_fusion PRIVATE -O2 -Wall -Wextra -pedantic)
target_link_libraries(matrix_sensor_fusion PRIVATE "${PACEVAL_RUNTIME}" dl pthread)
get_filename_component(PACEVAL_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
set_target_properties(matrix_sensor_fusion PROPERTIES BUILD_RPATH "${PACEVAL_DIR}")
enable_testing()
add_test(NAME genuine_original_sdk_small_spd COMMAND matrix_sensor_fusion ${CMAKE_SOURCE_DIR}/.. --n 16 --verify-only)
add_test(NAME genuine_original_sdk_sensor_fusion COMMAND matrix_sensor_fusion ${CMAKE_SOURCE_DIR}/.. --n 32 --verify-only)
add_test(NAME rejects_non_positive_definite_input COMMAND matrix_sensor_fusion ${CMAKE_SOURCE_DIR}/.. --n 8 --verify-only --indefinite-test)
set_tests_properties(rejects_non_positive_definite_input PROPERTIES WILL_FAIL TRUE)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME independent_numpy_audit COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/verify_results.py ${CMAKE_SOURCE_DIR}/..)
  add_test(NAME formula_execution_inventory COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/verify_inventory.py ${CMAKE_SOURCE_DIR}/..)
  add_test(NAME github_readme_math_and_links COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/check_readme_math.py ${CMAKE_SOURCE_DIR}/..)
endif()
