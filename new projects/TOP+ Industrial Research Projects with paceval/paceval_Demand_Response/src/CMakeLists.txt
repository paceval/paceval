cmake_minimum_required(VERSION 3.16)
project(paceval_demand_response_research LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This package is validated on Linux; use matching paceval runtime for any other OS")
endif()
if(NOT PACEVAL_RUNTIME AND DEFINED ENV{PACEVAL_LIB})
  set(PACEVAL_RUNTIME "$ENV{PACEVAL_LIB}")
endif()
if(NOT PACEVAL_RUNTIME OR NOT IS_ABSOLUTE "${PACEVAL_RUNTIME}" OR NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Set -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_runtime SHARED IMPORTED GLOBAL)
set_target_properties(paceval_runtime PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
# The official paceval C++ facade is compiled directly with the demonstrator.
add_executable(demand_response demand_response.cpp paceval_main.cpp)
target_include_directories(demand_response PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(demand_response PRIVATE paceval_runtime)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(demand_response PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME host_data_validation COMMAND demand_response --self-test)
  add_test(NAME sdk_all_rules COMMAND demand_response --rules 100 --days 14 --batch 37 --out "${CMAKE_CURRENT_BINARY_DIR}/test100")
  add_test(NAME sdk_single_batch COMMAND demand_response --rules 20 --days 4 --batch 1 --out "${CMAKE_CURRENT_BINARY_DIR}/test1")
  add_test(NAME sdk_csv_roundtrip COMMAND demand_response --input "${CMAKE_CURRENT_SOURCE_DIR}/../examples/hourly_trace.csv" --rules 4 --batch 64 --out "${CMAKE_CURRENT_BINARY_DIR}/roundtrip")
  add_test(NAME reject_invalid_hourly_csv COMMAND demand_response --input "${CMAKE_CURRENT_SOURCE_DIR}/../examples/invalid_trace.csv" --rules 4 --out "${CMAKE_CURRENT_BINARY_DIR}/bad")
  set_tests_properties(reject_invalid_hourly_csv PROPERTIES WILL_FAIL TRUE)
endif()
