cmake_minimum_required(VERSION 3.18)
project(paceval_wp25_robotic_grasp_wrench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PACEVAL_RUNTIME "" CACHE FILEPATH "Compatible original paceval Linux shared library, supplied separately")
if(NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Provide -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(robot_grasp_wrench_paceval robot_grasp_wrench_paceval.cpp paceval_main.cpp)
target_link_libraries(robot_grasp_wrench_paceval PRIVATE "${PACEVAL_RUNTIME}" dl pthread)
target_compile_options(robot_grasp_wrench_paceval PRIVATE -Wall -Wextra -Wpedantic)
get_filename_component(PACEVAL_LIB_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
set_target_properties(robot_grasp_wrench_paceval PROPERTIES BUILD_RPATH "${PACEVAL_LIB_DIR}")
enable_testing()
add_test(NAME original_sdk_operators_and_numeric_conversion COMMAND robot_grasp_wrench_paceval "${CMAKE_SOURCE_DIR}/.." --selftest)
add_test(NAME original_sdk_complete_grasp_graph_chain COMMAND robot_grasp_wrench_paceval "${CMAKE_SOURCE_DIR}/..")
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
 add_test(NAME archived_mcp_independent_reference COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/../validation/high_precision_reference.py" "${CMAKE_SOURCE_DIR}/..")
 add_test(NAME github_archive_math_preflight COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/../validation/check_project.py" "${CMAKE_SOURCE_DIR}/..")
endif()
