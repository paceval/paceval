cmake_minimum_required(VERSION 3.16)
project(paceval_carbon_capture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Supply -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_external SHARED IMPORTED)
set_target_properties(paceval_external PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(carbon_capture carbon_capture_main.cpp paceval_main.cpp)
target_include_directories(carbon_capture PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(carbon_capture PRIVATE paceval_external)
if(UNIX AND NOT APPLE)
 target_link_libraries(carbon_capture PRIVATE dl pthread)
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME sdk_scalar_batch_operators_conversions COMMAND carbon_capture --selftest)
add_test(NAME full_csv_to_paceval_to_csv COMMAND carbon_capture --run "${CMAKE_CURRENT_LIST_DIR}/../input/operating_points.csv" "${CMAKE_CURRENT_BINARY_DIR}/test_results")
add_test(NAME high_precision_process_reference COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/verify_high_precision.py" "${CMAKE_CURRENT_BINARY_DIR}/test_results" "${CMAKE_CURRENT_LIST_DIR}/../input/operating_points.csv")
set_tests_properties(high_precision_process_reference PROPERTIES DEPENDS full_csv_to_paceval_to_csv)
add_test(NAME archive_markdown_math_and_links COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/check_markdown.py" "${CMAKE_CURRENT_LIST_DIR}/..")
add_test(NAME markdown_checker_negative_tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/check_markdown.py" --selftest)
