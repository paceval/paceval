cmake_minimum_required(VERSION 3.16)
project(paceval_inverter_thermal LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT DEFINED PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Please supply -DPACEVAL_RUNTIME=/absolute/path/to/the/compatible/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_runtime SHARED IMPORTED GLOBAL)
set_target_properties(paceval_runtime PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(inverter_thermal inverter_twin.cpp paceval_main.cpp)
target_include_directories(inverter_thermal PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(inverter_thermal PRIVATE paceval_runtime dl)
target_compile_definitions(inverter_thermal PRIVATE PROJECT_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_options(inverter_thermal PRIVATE -Wall -Wextra -Wpedantic -O2)
set_target_properties(inverter_thermal PROPERTIES BUILD_RPATH "$<TARGET_FILE_DIR:paceval_runtime>")
enable_testing()
add_test(NAME official_converter_schema COMMAND inverter_thermal --test-conversion)
add_test(NAME operator_logical_relational_semantics COMMAND inverter_thermal --test-operators)
add_test(NAME scalar_and_multigraph_batch_layout COMMAND inverter_thermal --test-layout)
add_test(NAME invalid_domain_input_rejection COMMAND inverter_thermal --test-invalid)
add_test(NAME functional_synthetic_sweep COMMAND inverter_thermal --run --count 512 --out "${CMAKE_BINARY_DIR}/test_sweep")
add_test(NAME independent_full_output_audit COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/verify_reference.py" --results "${CMAKE_BINARY_DIR}/test_sweep" --small)
set_tests_properties(independent_full_output_audit PROPERTIES DEPENDS functional_synthetic_sweep)
add_test(NAME readme_links_math_and_hero COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/check_readme_math.py")
add_test(NAME readme_negative_link_regression COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/check_readme_math.py" --self-test)
add_test(NAME archive_checksum_manifest COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/verify_manifest.py")
