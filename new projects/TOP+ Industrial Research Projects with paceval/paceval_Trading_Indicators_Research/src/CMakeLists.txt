cmake_minimum_required(VERSION 3.16)
project(paceval_interchangeable_trading_indicators LANGUAGES CXX)
# Official API facade compiled as source. Compatible runtime is an external
# binary supplied explicitly at configure time; never redistribute it here.
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Specify -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(trading_indicators trading_indicators.cpp paceval_main.cpp)
target_compile_features(trading_indicators PRIVATE cxx_std_17)
target_include_directories(trading_indicators PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(trading_indicators PRIVATE "${PACEVAL_RUNTIME}")
if(MSVC)
 target_compile_options(trading_indicators PRIVATE /W4)
else()
 target_compile_options(trading_indicators PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(UNIX AND NOT APPLE)
 get_filename_component(RUNTIME_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
 set_target_properties(trading_indicators PROPERTIES BUILD_RPATH "${RUNTIME_DIR}")
endif()
enable_testing()
add_test(NAME sdk_indicator_integration COMMAND trading_indicators --out "${CMAKE_CURRENT_BINARY_DIR}/test_outputs")
