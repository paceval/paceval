cmake_minimum_required(VERSION 3.16)
project(paceval_counterparty_exposure LANGUAGES CXX)
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Specify -DPACEVAL_RUNTIME=/path/to/compatible/libpaceval_linux_sharedLIB.so")
endif()
add_executable(counterparty_exposure counterparty_exposure.cpp paceval_main.cpp)
target_compile_features(counterparty_exposure PRIVATE cxx_std_17)
target_include_directories(counterparty_exposure PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(counterparty_exposure PRIVATE "${PACEVAL_RUNTIME}")
if(NOT MSVC)
 target_compile_options(counterparty_exposure PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(UNIX AND NOT APPLE)
 get_filename_component(RUNTIME_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
 set_target_properties(counterparty_exposure PROPERTIES BUILD_RPATH "${RUNTIME_DIR}")
endif()
enable_testing()
add_test(NAME counterparty_sdk_integration COMMAND counterparty_exposure --updates 2500 --out "${CMAKE_CURRENT_BINARY_DIR}/test_outputs")
