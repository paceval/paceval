cmake_minimum_required(VERSION 3.16)
project(paceval_fermi_2fle_real_catalog LANGUAGES CXX)
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Pass -DPACEVAL_RUNTIME=/absolute/path/to/compatible/libpaceval_linux_sharedLIB.so")
endif()
find_library(CFITSIO_LIBRARY cfitsio REQUIRED)
add_executable(fermi_real_catalog fermi_real_catalog.cpp paceval_main.cpp)
target_compile_features(fermi_real_catalog PRIVATE cxx_std_17)
target_include_directories(fermi_real_catalog PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(fermi_real_catalog PRIVATE "${PACEVAL_RUNTIME}" "${CFITSIO_LIBRARY}")
if(UNIX AND NOT APPLE)
 get_filename_component(PACEVAL_LIB_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
 set_target_properties(fermi_real_catalog PROPERTIES BUILD_RPATH "${PACEVAL_LIB_DIR}")
endif()
target_compile_options(fermi_real_catalog PRIVATE -O2 -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME original_fits_real_sdk COMMAND fermi_real_catalog --fits "${CMAKE_CURRENT_SOURCE_DIR}/../data/2FLE_psc_v8.fits" --legacy "${CMAKE_CURRENT_SOURCE_DIR}/../data/legacy_60_report_metadata.csv" --out "${CMAKE_CURRENT_BINARY_DIR}/ctest_run" --limit 12)
