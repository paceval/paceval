cmake_minimum_required(VERSION 3.16)
project(paceval_nonlinear_lyapunov LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Specify -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_external SHARED IMPORTED)
set_target_properties(paceval_external PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(nonlinear_lyapunov nonlinear_control_main.cpp paceval_main.cpp)
target_include_directories(nonlinear_lyapunov PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(nonlinear_lyapunov PRIVATE paceval_external)
if(UNIX AND NOT APPLE)
 target_link_libraries(nonlinear_lyapunov PRIVATE dl pthread)
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
enable_testing()
add_test(NAME sdk_scalar_batch_operators_conversion COMMAND nonlinear_lyapunov --selftest)
add_test(NAME csv_to_real_paceval_to_csv COMMAND nonlinear_lyapunov --run "${CMAKE_CURRENT_LIST_DIR}/../input/candidates.csv" "${CMAKE_CURRENT_BINARY_DIR}/test_results")
add_test(NAME independent_high_precision COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/verify_high_precision.py" "${CMAKE_CURRENT_BINARY_DIR}/test_results" "${CMAKE_CURRENT_LIST_DIR}/../input/candidates.csv")
set_tests_properties(independent_high_precision PROPERTIES DEPENDS csv_to_real_paceval_to_csv)
add_test(NAME analytic_lyapunov_certificate COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/verify_certificates.py" "${CMAKE_CURRENT_BINARY_DIR}/test_results" "${CMAKE_CURRENT_LIST_DIR}/../input/candidates.csv")
set_tests_properties(analytic_lyapunov_certificate PROPERTIES DEPENDS csv_to_real_paceval_to_csv)
add_test(NAME archive_markdown_math_links COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/check_markdown.py" "${CMAKE_CURRENT_LIST_DIR}/..")
add_test(NAME markdown_negative_regressions COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/check_markdown.py" --selftest)
add_test(NAME independently_integrated_closed_loop COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/verify_trajectories.py" "${CMAKE_CURRENT_BINARY_DIR}/test_results" "${CMAKE_CURRENT_LIST_DIR}/../input/candidates.csv")
set_tests_properties(independently_integrated_closed_loop PROPERTIES DEPENDS csv_to_real_paceval_to_csv)
add_test(NAME deterministic_input_generation COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/generate_input.py" --check "${CMAKE_CURRENT_LIST_DIR}/../input/candidates.csv")
add_test(NAME scientific_release_file_preflight COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/check_release.py")
