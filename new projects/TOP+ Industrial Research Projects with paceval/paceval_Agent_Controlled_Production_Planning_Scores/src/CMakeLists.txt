cmake_minimum_required(VERSION 3.16)
project(paceval_agent_controlled_production_planning LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PACEVAL_RUNTIME "" CACHE FILEPATH "Absolute path to the separately supplied paceval Linux shared library")
if(NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Specify -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(production_planning production_planning.cpp paceval_main.cpp)
target_compile_options(production_planning PRIVATE -O2 -Wall -Wextra -Wpedantic)
target_link_libraries(production_planning PRIVATE "${PACEVAL_RUNTIME}" dl pthread)
get_filename_component(PACEVAL_LIB_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
set_target_properties(production_planning PROPERTIES BUILD_RPATH "${PACEVAL_LIB_DIR}")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_test(NAME original_paceval_sdk_execution COMMAND production_planning --out "${CMAKE_BINARY_DIR}/test_run" --fast)
set_tests_properties(original_paceval_sdk_execution PROPERTIES FIXTURES_SETUP sdk_outputs)
add_test(NAME independent_schedule_numeric_audit COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_results.py" --root "${CMAKE_BINARY_DIR}/test_run")
set_tests_properties(independent_schedule_numeric_audit PROPERTIES FIXTURES_REQUIRED sdk_outputs)
add_test(NAME github_readme_math_and_links COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_readme_math.py")
add_test(NAME rejection_of_unvalidated_agent_requests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_negative_agent_input.py" "$<TARGET_FILE:production_planning>")
