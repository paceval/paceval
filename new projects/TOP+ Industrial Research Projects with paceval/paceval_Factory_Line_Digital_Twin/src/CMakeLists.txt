cmake_minimum_required(VERSION 3.16)
project(paceval_factory_line_twin LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PACEVAL_RUNTIME "" CACHE FILEPATH "Path to original paceval Linux .so")
if(NOT EXISTS "${PACEVAL_RUNTIME}")
    message(FATAL_ERROR "Supply the external original paceval shared library with -DPACEVAL_RUNTIME=/absolute/path/libpaceval_linux_sharedLIB.so")
endif()
add_executable(factory_line_twin factory_line_twin.cpp paceval_main.cpp)
target_link_libraries(factory_line_twin PRIVATE "${PACEVAL_RUNTIME}" dl pthread)
target_compile_options(factory_line_twin PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(factory_line_twin PROPERTIES BUILD_RPATH "$<TARGET_FILE_DIR:factory_line_twin>;${CMAKE_CURRENT_BINARY_DIR}")
get_filename_component(PACEVAL_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
set_target_properties(factory_line_twin PROPERTIES BUILD_RPATH "${PACEVAL_DIR}")
enable_testing()
add_test(NAME original_paceval_integration COMMAND factory_line_twin --out "${CMAKE_CURRENT_BINARY_DIR}/ctest" --jobs 80 --variants 3 --surface 9)
