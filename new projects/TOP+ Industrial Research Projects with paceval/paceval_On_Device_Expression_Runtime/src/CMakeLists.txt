cmake_minimum_required(VERSION 3.16)
project(paceval_on_device_expressions LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# paceval_main.cpp is the official SDK forwarding/interface layer.
# The architecture-matched binary mathematical runtime is supplied separately.
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This release provides Linux validation; adapt runtime for other targets")
endif()
if(NOT PACEVAL_RUNTIME AND DEFINED ENV{PACEVAL_LIB})
  set(PACEVAL_RUNTIME "$ENV{PACEVAL_LIB}")
endif()
if(NOT PACEVAL_RUNTIME OR NOT IS_ABSOLUTE "${PACEVAL_RUNTIME}" OR NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Pass -DPACEVAL_RUNTIME=/absolute/path/to/architecture-matched/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_binary SHARED IMPORTED GLOBAL)
set_target_properties(paceval_binary PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(on_device_runtime on_device_runtime.cpp paceval_main.cpp)
target_include_directories(on_device_runtime PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(on_device_runtime PRIVATE paceval_binary)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(on_device_runtime PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
 add_test(NAME reference_self_test COMMAND on_device_runtime --self-test)
 add_test(NAME real_runtime_smoke COMMAND on_device_runtime --models 12 --samples 10 --batch 3 --out "${CMAKE_CURRENT_BINARY_DIR}/smoke")
 add_test(NAME real_runtime_versioning COMMAND on_device_runtime --models 32 --samples 20 --batch 7 --out "${CMAKE_CURRENT_BINARY_DIR}/versioning")
endif()
