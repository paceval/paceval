cmake_minimum_required(VERSION 3.16)
project(paceval_distillation_vle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Set -DPACEVAL_RUNTIME=/absolute/path/to/compatible/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_runtime SHARED IMPORTED GLOBAL)
set_target_properties(paceval_runtime PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(distillation_kernels distillation_kernels.cpp paceval_main.cpp)
target_link_libraries(distillation_kernels PRIVATE paceval_runtime dl)
set_target_properties(distillation_kernels PROPERTIES BUILD_RPATH "$<TARGET_FILE_DIR:paceval_runtime>")
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(distillation_kernels PRIVATE -O2)
endif()
enable_testing()
add_test(NAME original_sdk_operator_csv_smoke COMMAND distillation_kernels --selftest)
add_test(NAME original_sdk_vle_small COMMAND distillation_kernels --run 64 16 "${CMAKE_BINARY_DIR}/ctest_small")
add_test(NAME original_sdk_vle_extended COMMAND distillation_kernels --run 512 32 "${CMAKE_BINARY_DIR}/ctest_medium")
set_tests_properties(original_sdk_operator_csv_smoke original_sdk_vle_small original_sdk_vle_extended PROPERTIES ENVIRONMENT "LD_LIBRARY_PATH=$ENV{LD_LIBRARY_PATH}")

find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME independent_high_precision_and_stage_reference COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../validation/validate_results.py" "${CMAKE_CURRENT_SOURCE_DIR}/../results")
add_test(NAME archive_wide_github_markdown_math COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_archive_markdown.py")
add_test(NAME peer_review_publication_integrity COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_publication.py")
add_test(NAME machine_readable_execution_evidence COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_archive_evidence.py")
add_test(NAME sha256_distribution_manifest COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_manifest.py")
