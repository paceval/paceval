cmake_minimum_required(VERSION 3.18)
project(paceval_decision_tree_research LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PACEVAL_RUNTIME "" CACHE FILEPATH "Absolute path to original paceval Linux shared library")
if(NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "A compatible original paceval shared library is required. Specify -DPACEVAL_RUNTIME=...")
endif()
add_executable(decision_tree decision_tree.cpp paceval_main.cpp)
target_compile_options(decision_tree PRIVATE -O2 -Wall -Wextra -Wpedantic)
target_link_libraries(decision_tree PRIVATE "${PACEVAL_RUNTIME}" dl pthread)
get_filename_component(PACEVAL_LIB_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
set_target_properties(decision_tree PROPERTIES BUILD_RPATH "${PACEVAL_LIB_DIR}")
enable_testing()
add_test(NAME real_paceval_end_to_end COMMAND decision_tree ${CMAKE_SOURCE_DIR}/..)
add_test(NAME invalid_range_is_rejected COMMAND decision_tree ${CMAKE_SOURCE_DIR}/.. --invalid)
set_tests_properties(invalid_range_is_rejected PROPERTIES WILL_FAIL TRUE)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
 add_test(NAME independent_reference COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/independent_reference.py ${CMAKE_SOURCE_DIR}/..)
 add_test(NAME graph_coverage COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/verify_inventory.py ${CMAKE_SOURCE_DIR}/..)
 add_test(NAME readme_math_and_links COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/check_readme_math.py ${CMAKE_SOURCE_DIR}/..)
endif()

# Complete 20-level structured tree, all 1,048,576 paths tested through original paceval API.
add_executable(depth20_exhaustive depth20_exhaustive.cpp paceval_main.cpp)
target_compile_options(depth20_exhaustive PRIVATE -O2 -Wall -Wextra -Wpedantic)
target_link_libraries(depth20_exhaustive PRIVATE "${PACEVAL_RUNTIME}" dl pthread)
set_target_properties(depth20_exhaustive PROPERTIES BUILD_RPATH "${PACEVAL_LIB_DIR}")
add_test(NAME depth20_all_1048576_paths COMMAND depth20_exhaustive ${CMAKE_SOURCE_DIR}/..)

if(Python3_Interpreter_FOUND)
 add_test(NAME depth20_independent_80digit_reference COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/verify_depth20_precision.py ${CMAKE_SOURCE_DIR}/..)
endif()
