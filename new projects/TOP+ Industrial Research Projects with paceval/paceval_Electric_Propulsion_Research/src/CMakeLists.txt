cmake_minimum_required(VERSION 3.16)
project(paceval_electric_propulsion LANGUAGES CXX)
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Supply -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(electric_propulsion electric_propulsion.cpp paceval_main.cpp)
target_compile_features(electric_propulsion PRIVATE cxx_std_17)
target_include_directories(electric_propulsion PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(electric_propulsion PRIVATE "${PACEVAL_RUNTIME}")
if(NOT MSVC)
 target_compile_options(electric_propulsion PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(UNIX AND NOT APPLE)
 get_filename_component(RUNTIME_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
 set_target_properties(electric_propulsion PROPERTIES BUILD_RPATH "${RUNTIME_DIR}")
endif()
enable_testing()
add_test(NAME electric_propulsion_sdk_integration COMMAND electric_propulsion --updates 1000 --out "${CMAKE_CURRENT_BINARY_DIR}/ctest_outputs")
