cmake_minimum_required(VERSION 3.16)
project(paceval_alm_stress LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Specify -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_runtime SHARED IMPORTED)
set_target_properties(paceval_runtime PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(alm_stress alm_stress.cpp paceval_main.cpp)
target_include_directories(alm_stress PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(alm_stress PRIVATE paceval_runtime)
if(UNIX AND NOT APPLE)
 target_link_libraries(alm_stress PRIVATE dl pthread)
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
enable_testing()
add_test(NAME real_paceval_sdk_conversion_and_operator_semantics COMMAND alm_stress --selftest)
add_test(NAME cashflow_csv_paceval_batches COMMAND alm_stress --run "${CMAKE_CURRENT_LIST_DIR}/../input/cashflows_scenarios.csv" "${CMAKE_CURRENT_BINARY_DIR}/test_results")
add_test(NAME independent_mpmath_financial_validation COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/verify_high_precision.py" "${CMAKE_CURRENT_LIST_DIR}/../input/cashflows_scenarios.csv" "${CMAKE_CURRENT_BINARY_DIR}/test_results")
set_tests_properties(independent_mpmath_financial_validation PROPERTIES DEPENDS cashflow_csv_paceval_batches)
add_test(NAME independent_curve_and_shock_invariants COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/verify_invariants.py" "${CMAKE_CURRENT_BINARY_DIR}/test_results")
set_tests_properties(independent_curve_and_shock_invariants PROPERTIES DEPENDS cashflow_csv_paceval_batches)
add_test(NAME github_markdown_math_links COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/check_markdown.py" "${CMAKE_CURRENT_LIST_DIR}/..")
add_test(NAME github_markdown_negative_tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/check_markdown.py" --selftest)
add_test(NAME deterministic_input_generator COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/generate_input.py" --check)
add_test(NAME scientific_release_preflight COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/check_release.py")
set_tests_properties(scientific_release_preflight PROPERTIES DEPENDS independent_mpmath_financial_validation)
