cmake_minimum_required(VERSION 3.16)
project(paceval_composite_cure LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Supply -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_external SHARED IMPORTED)
set_target_properties(paceval_external PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(composite_cure composite_cure_main.cpp paceval_main.cpp)
target_include_directories(composite_cure PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(composite_cure PRIVATE paceval_external)
if(UNIX AND NOT APPLE)
 target_link_libraries(composite_cure PRIVATE dl pthread)
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
enable_testing()
add_test(NAME sdk_scalar_batch_operators_conversion COMMAND composite_cure --selftest)
add_test(NAME full_csv_paceval_pipeline COMMAND composite_cure --run "${CMAKE_CURRENT_LIST_DIR}/../input/recipes.csv" "${CMAKE_CURRENT_BINARY_DIR}/ctest_results")
add_test(NAME high_precision_independent_reference COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/verify_high_precision.py" "${CMAKE_CURRENT_BINARY_DIR}/ctest_results")
set_tests_properties(high_precision_independent_reference PROPERTIES DEPENDS full_csv_paceval_pipeline)
add_test(NAME archive_markdown_math_links COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/check_markdown.py" "${CMAKE_CURRENT_LIST_DIR}/..")
add_test(NAME markdown_negative_regression COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/check_markdown.py" --selftest)

add_test(NAME independent_full_time_trajectory COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/verify_trajectory.py" "${CMAKE_CURRENT_BINARY_DIR}/ctest_results")
set_tests_properties(independent_full_time_trajectory PROPERTIES DEPENDS full_csv_paceval_pipeline)
