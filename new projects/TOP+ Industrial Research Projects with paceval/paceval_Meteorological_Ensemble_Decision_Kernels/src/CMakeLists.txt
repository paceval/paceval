cmake_minimum_required(VERSION 3.16)
project(paceval_weather_ensemble_kernels LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Supply -DPACEVAL_RUNTIME=/absolute/path/to/compatible/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_runtime SHARED IMPORTED GLOBAL)
set_target_properties(paceval_runtime PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(weather_kernels weather_kernels.cpp paceval_main.cpp)
target_include_directories(weather_kernels PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(weather_kernels PRIVATE paceval_runtime dl)
target_compile_options(weather_kernels PRIVATE -O2 -Wall -Wextra -Wpedantic)
set_target_properties(weather_kernels PROPERTIES BUILD_RPATH "$<TARGET_FILE_DIR:paceval_runtime>")
enable_testing()
add_test(NAME paceval_numeric_csv_conversion COMMAND weather_kernels --test-conversion)
add_test(NAME paceval_combined_operator_semantics COMMAND weather_kernels --test-operators)
add_test(NAME paceval_scalar_multigraph_batch_layout COMMAND weather_kernels --test-layout)
add_test(NAME negative_input_domain_cases COMMAND weather_kernels --test-invalid)
add_test(NAME deterministic_synthetic_sample COMMAND weather_kernels --run --sites 24 --out "${CMAKE_BINARY_DIR}/weather_sample")
add_test(NAME independent_python_sample_reference COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/../validation/verify_weather_reference.py" "${CMAKE_BINARY_DIR}/weather_sample")
set_tests_properties(independent_python_sample_reference PROPERTIES DEPENDS deterministic_synthetic_sample)
add_test(NAME full_archived_high_precision_reference COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/../validation/verify_weather_reference.py" "${CMAKE_CURRENT_SOURCE_DIR}/../results")
add_test(NAME published_graph_counts_and_reliability COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/check_archive_evidence.py")
add_test(NAME github_readme_links_math_and_hero COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/check_readme.py")
add_test(NAME github_readme_negative_link_test COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/check_readme.py" --self-test)
add_test(NAME whitepaper_and_figure_integrity COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/check_publication.py")
add_test(NAME sha256_distribution_integrity COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/verify_manifest.py")
