cmake_minimum_required(VERSION 3.16)
project(paceval_intraday_risk LANGUAGES CXX)
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Specify -DPACEVAL_RUNTIME=/absolute/path/to/compatible/libpaceval_linux_sharedLIB.so")
endif()
add_executable(intraday_risk intraday_risk.cpp paceval_main.cpp)
target_compile_features(intraday_risk PRIVATE cxx_std_17)
target_include_directories(intraday_risk PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(intraday_risk PRIVATE "${PACEVAL_RUNTIME}")
if(NOT MSVC)
 target_compile_options(intraday_risk PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(UNIX AND NOT APPLE)
 get_filename_component(RUNTIME_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
 set_target_properties(intraday_risk PROPERTIES BUILD_RPATH "${RUNTIME_DIR}")
endif()
enable_testing()
add_test(NAME intraday_sdk_integration COMMAND intraday_risk --updates 2500 --out "${CMAKE_CURRENT_BINARY_DIR}/test_outputs")
