cmake_minimum_required(VERSION 3.16)
project(paceval_orbit_parameter_sweeps LANGUAGES CXX)
# SDK interface comes from its original public C/C++ source pair. The matching
# architecture-specific external Linux runtime must be supplied explicitly.
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Supply -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(orbit_sweeps orbit_sweeps.cpp paceval_main.cpp)
target_compile_features(orbit_sweeps PRIVATE cxx_std_17)
target_include_directories(orbit_sweeps PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(orbit_sweeps PRIVATE "${PACEVAL_RUNTIME}")
if(MSVC)
  target_compile_options(orbit_sweeps PRIVATE /W4)
else()
  target_compile_options(orbit_sweeps PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(UNIX AND NOT APPLE)
  get_filename_component(PACEVAL_LIB_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
  set_target_properties(orbit_sweeps PROPERTIES BUILD_RPATH "${PACEVAL_LIB_DIR}")
endif()
enable_testing()
add_test(NAME orbit_propagation_and_sweeps COMMAND orbit_sweeps --out "${CMAKE_CURRENT_BINARY_DIR}/test_outputs")
