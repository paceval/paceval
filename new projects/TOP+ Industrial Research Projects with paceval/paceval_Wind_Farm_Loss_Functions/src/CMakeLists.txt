cmake_minimum_required(VERSION 3.16)
project(paceval_wind_farm_loss_functions LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Linux research demonstration; supply an appropriate paceval runtime for other platforms")
endif()
if(NOT PACEVAL_RUNTIME AND DEFINED ENV{PACEVAL_LIB})
  set(PACEVAL_RUNTIME "$ENV{PACEVAL_LIB}")
endif()
if(NOT PACEVAL_RUNTIME OR NOT IS_ABSOLUTE "${PACEVAL_RUNTIME}" OR NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Supply -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_runtime SHARED IMPORTED GLOBAL)
set_target_properties(paceval_runtime PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(wind_farm wind_farm.cpp paceval_main.cpp)
target_include_directories(wind_farm PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(wind_farm PRIVATE paceval_runtime)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(wind_farm PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME native_validation COMMAND wind_farm --self-test)
  add_test(NAME sdk_100_models COMMAND wind_farm --variants 100 --batch 128 --out "${CMAKE_CURRENT_BINARY_DIR}/full")
  add_test(NAME sdk_scalar_batch COMMAND wind_farm --variants 4 --batch 1 --out "${CMAKE_CURRENT_BINARY_DIR}/scalar")
  add_test(NAME sdk_csv_roundtrip COMMAND wind_farm --variants 4 --batch 37 --wind-input "${CMAKE_CURRENT_SOURCE_DIR}/../examples/wind_conditions.csv" --out "${CMAKE_CURRENT_BINARY_DIR}/csv")
  add_test(NAME reject_invalid_csv COMMAND wind_farm --variants 4 --wind-input "${CMAKE_CURRENT_SOURCE_DIR}/../examples/invalid_wind_conditions.csv" --out "${CMAKE_CURRENT_BINARY_DIR}/invalid")
  set_tests_properties(reject_invalid_csv PROPERTIES WILL_FAIL TRUE)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME independent_high_precision
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/high_precision_reference.py" "${CMAKE_CURRENT_BINARY_DIR}/full")
    set_tests_properties(independent_high_precision PROPERTIES DEPENDS sdk_100_models)
  endif()
endif()
