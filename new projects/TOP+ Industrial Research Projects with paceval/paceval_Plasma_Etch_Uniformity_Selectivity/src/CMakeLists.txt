cmake_minimum_required(VERSION 3.16)
project(wp06_plasma_etch LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Set -DPACEVAL_RUNTIME=/absolute/path/to/original/compatible/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_runtime SHARED IMPORTED GLOBAL)
set_target_properties(paceval_runtime PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(plasma_etch plasma_etch.cpp paceval_main.cpp)
target_include_directories(plasma_etch PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(plasma_etch PRIVATE paceval_runtime dl)
target_compile_options(plasma_etch PRIVATE -O2 -Wall -Wextra -Wpedantic)
set_target_properties(plasma_etch PROPERTIES BUILD_RPATH "$<TARGET_FILE_DIR:paceval_runtime>")
enable_testing()
add_test(NAME numeric_csv_conversion COMMAND plasma_etch --test-conversion)
add_test(NAME logical_relational_operator_semantics COMMAND plasma_etch --test-operators)
add_test(NAME scalar_multigraph_layout COMMAND plasma_etch --test-layout)
add_test(NAME invalid_domain_rejection COMMAND plasma_etch --test-invalid)
add_test(NAME full_synthetic_small_run COMMAND plasma_etch --run --recipes 2 --radial 8 --angular 16 --out "${CMAKE_BINARY_DIR}/test_wp06")
add_test(NAME independent_small_reference COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/verify_reference.py" "${CMAKE_BINARY_DIR}/test_wp06" --small)
set_tests_properties(independent_small_reference PROPERTIES DEPENDS full_synthetic_small_run)
add_test(NAME readme_figure_and_math_links COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/check_readme_math.py")
add_test(NAME readme_negative_regression COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/check_readme_math.py" --self-test)
add_test(NAME full_independent_ledger_manifest COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/verify_manifest.py")
