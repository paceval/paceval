cmake_minimum_required(VERSION 3.16)
project(paceval_optical_film LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Specify -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_runtime SHARED IMPORTED)
set_target_properties(paceval_runtime PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(optical_film optical_film_main.cpp paceval_main.cpp)
target_include_directories(optical_film PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(optical_film PRIVATE paceval_runtime)
if(UNIX AND NOT APPLE)
 target_link_libraries(optical_film PRIVATE dl pthread)
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
enable_testing()
add_test(NAME real_sdk_operators_and_conversion COMMAND optical_film --selftest)
add_test(NAME csv_paceval_multigraph_inverse COMMAND optical_film --run "${CMAKE_CURRENT_LIST_DIR}/../input/spectral_samples.csv" "${CMAKE_CURRENT_BINARY_DIR}/test_results")
add_test(NAME independent_high_precision COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/verify_high_precision.py" "${CMAKE_CURRENT_BINARY_DIR}/test_results" "${CMAKE_CURRENT_LIST_DIR}/../input/spectral_samples.csv")
set_tests_properties(independent_high_precision PROPERTIES DEPENDS csv_paceval_multigraph_inverse)
add_test(NAME independent_inversion_and_alias COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/verify_inversion.py" "${CMAKE_CURRENT_BINARY_DIR}/test_results")
set_tests_properties(independent_inversion_and_alias PROPERTIES DEPENDS csv_paceval_multigraph_inverse)
add_test(NAME github_markdown_math_and_links COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/check_markdown.py" "${CMAKE_CURRENT_LIST_DIR}/..")
add_test(NAME markdown_negative_regressions COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/check_markdown.py" --selftest)
add_test(NAME deterministic_input_generation COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/verify_generator.py")
add_test(NAME scientific_release_file_preflight COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/check_release.py")
