cmake_minimum_required(VERSION 3.16)
project(paceval_monte_carlo_pricing LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# The paceval SDK's C++ interface is compiled with this program. The actual
# architecture-specific paceval runtime is an external dependency: no .so file
# is bundled in the public repository.
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This demonstrator is tested on Linux; porting needs a compatible runtime")
endif()
if(NOT PACEVAL_RUNTIME AND DEFINED ENV{PACEVAL_LIB})
  set(PACEVAL_RUNTIME "$ENV{PACEVAL_LIB}")
endif()
if(NOT PACEVAL_RUNTIME OR NOT IS_ABSOLUTE "${PACEVAL_RUNTIME}" OR NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Supply -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_runtime SHARED IMPORTED GLOBAL)
set_target_properties(paceval_runtime PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(monte_carlo_pricing monte_carlo_pricing.cpp paceval_main.cpp)
target_include_directories(monte_carlo_pricing PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(monte_carlo_pricing PRIVATE paceval_runtime)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(monte_carlo_pricing PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME native_reference_selftest COMMAND monte_carlo_pricing --self-test)
  add_test(NAME runtime_payoff_validation COMMAND monte_carlo_pricing --paths 2048 --contracts 20 --steps 8 --batch 37 --out "${CMAKE_CURRENT_BINARY_DIR}/smoke")
  add_test(NAME runtime_versioning_and_tinc COMMAND monte_carlo_pricing --paths 4096 --contracts 25 --steps 16 --batch 128 --out "${CMAKE_CURRENT_BINARY_DIR}/versioning")
endif()
