cmake_minimum_required(VERSION 3.18)
project(paceval_industrial_matrix_precision LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PACEVAL_RUNTIME "" CACHE FILEPATH "Absolute path to an ABI-compatible original Linux paceval shared library")
if(NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Original paceval Linux .so unavailable. Pass -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(matrix_precision_paceval matrix_precision_paceval.cpp paceval_main.cpp)
target_link_libraries(matrix_precision_paceval PRIVATE "${PACEVAL_RUNTIME}" dl pthread)
target_compile_options(matrix_precision_paceval PRIVATE -Wall -Wextra -Wpedantic -O2)
get_filename_component(PACEVAL_LIB_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
set_target_properties(matrix_precision_paceval PROPERTIES BUILD_RPATH "${PACEVAL_LIB_DIR}")
enable_testing()
add_test(NAME original_sdk_matrix_precision COMMAND matrix_precision_paceval "${CMAKE_SOURCE_DIR}/..")
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME high_precision_mcp_record_audit COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/high_precision_reference.py "${CMAKE_SOURCE_DIR}/..")
  add_test(NAME github_readme_links_and_math COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/check_readme_math.py "${CMAKE_SOURCE_DIR}/..")
endif()
