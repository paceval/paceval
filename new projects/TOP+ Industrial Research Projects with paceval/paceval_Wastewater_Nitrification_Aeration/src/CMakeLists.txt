cmake_minimum_required(VERSION 3.18)
project(paceval_wp18_wastewater_kernels LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PACEVAL_RUNTIME "" CACHE FILEPATH "Original ABI-compatible paceval Linux shared library (not redistributed)")
if(NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Provide -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(wastewater_paceval wastewater_paceval.cpp paceval_main.cpp)
target_link_libraries(wastewater_paceval PRIVATE "${PACEVAL_RUNTIME}" dl pthread)
target_compile_options(wastewater_paceval PRIVATE -Wall -Wextra -Wpedantic)
get_filename_component(PACEVAL_LIB_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
set_target_properties(wastewater_paceval PROPERTIES BUILD_RPATH "${PACEVAL_LIB_DIR}")
enable_testing()
add_test(NAME original_sdk_conversion_and_operator_selftest COMMAND wastewater_paceval "${CMAKE_SOURCE_DIR}/.." --selftest)
add_test(NAME original_sdk_batch_and_nitrogen_validation COMMAND wastewater_paceval "${CMAKE_SOURCE_DIR}/..")
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
 add_test(NAME independent_archived_mcp_reference COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/../validation/high_precision_reference.py" "${CMAKE_SOURCE_DIR}/..")
 add_test(NAME archived_operator_semantics COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/../validation/check_operator_results.py" "${CMAKE_SOURCE_DIR}/..")
 add_test(NAME project_and_markdown_preflight COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/../validation/check_project.py" "${CMAKE_SOURCE_DIR}/..")
endif()
