cmake_minimum_required(VERSION 3.16)
project(paceval_wp031_trajectory_costs LANGUAGES CXX)
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Supply -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(trajectory_costs trajectory_costs.cpp paceval_main.cpp)
target_compile_features(trajectory_costs PRIVATE cxx_std_17)
target_include_directories(trajectory_costs PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(trajectory_costs PRIVATE "${PACEVAL_RUNTIME}")
if(UNIX AND NOT APPLE)
 get_filename_component(RUNTIME_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
 set_target_properties(trajectory_costs PROPERTIES BUILD_RPATH "${RUNTIME_DIR}")
endif()
enable_testing()
add_test(NAME wp031_original_sdk COMMAND trajectory_costs --out "${CMAKE_CURRENT_BINARY_DIR}/test_data")
