cmake_minimum_required(VERSION 3.16)
project(paceval_agent_simulations LANGUAGES CXX)
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Provide -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(agent_simulations agent_simulations.cpp paceval_main.cpp)
target_compile_features(agent_simulations PRIVATE cxx_std_17)
target_compile_options(agent_simulations PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(agent_simulations PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(agent_simulations PRIVATE "${PACEVAL_RUNTIME}")
get_filename_component(PACEVAL_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
set_target_properties(agent_simulations PROPERTIES BUILD_RPATH "${PACEVAL_DIR}")
enable_testing()
add_test(NAME sdk_orchestration COMMAND agent_simulations --out "${CMAKE_CURRENT_BINARY_DIR}/ctest_examples")
