cmake_minimum_required(VERSION 3.16)
project(paceval_advanced_chip_packaging LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Set -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval*.so")
endif()
add_library(paceval_runtime SHARED IMPORTED)
set_target_properties(paceval_runtime PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(chip_warpage warpage_main.cpp paceval_main.cpp)
target_include_directories(chip_warpage PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(chip_warpage PRIVATE paceval_runtime)
if(UNIX AND NOT APPLE)
  target_link_libraries(chip_warpage PRIVATE dl pthread)
endif()
enable_testing()
add_test(NAME sdk_graphs_and_operator_tests COMMAND chip_warpage --selftest)
add_test(NAME readme_math_and_links COMMAND python3 "${CMAKE_CURRENT_LIST_DIR}/check_readme_math.py" "${CMAKE_CURRENT_LIST_DIR}/../README.md")
add_test(NAME readme_checker_regressions COMMAND python3 "${CMAKE_CURRENT_LIST_DIR}/check_readme_math.py" --selftest)
add_test(NAME csv_to_paceval_pipeline COMMAND chip_warpage --run "${CMAKE_CURRENT_LIST_DIR}/../input/scenarios.csv" "${CMAKE_CURRENT_BINARY_DIR}/integration_results")
add_test(NAME independent_high_precision_validation COMMAND python3 "${CMAKE_CURRENT_LIST_DIR}/../scripts/verify_high_precision.py" "${CMAKE_CURRENT_BINARY_DIR}/integration_results")
set_tests_properties(independent_high_precision_validation PROPERTIES DEPENDS csv_to_paceval_pipeline)
