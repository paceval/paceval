cmake_minimum_required(VERSION 3.16)
project(paceval_gravitational_inversion LANGUAGES CXX)
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Provide -DPACEVAL_RUNTIME=/absolute/path/to/compatible/libpaceval_linux_sharedLIB.so")
endif()
add_executable(gravity_inversion gravity_inversion.cpp paceval_main.cpp)
target_compile_features(gravity_inversion PRIVATE cxx_std_17)
target_include_directories(gravity_inversion PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(gravity_inversion PRIVATE "${PACEVAL_RUNTIME}")
target_compile_options(gravity_inversion PRIVATE -Wall -Wextra -Wpedantic)
if(UNIX AND NOT APPLE)
 get_filename_component(PACEVAL_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
 set_target_properties(gravity_inversion PROPERTIES BUILD_RPATH "${PACEVAL_DIR}")
endif()
enable_testing()
add_test(NAME wp033_sdk_and_inversion COMMAND gravity_inversion --out "${CMAKE_CURRENT_BINARY_DIR}/ctest_examples")
