cmake_minimum_required(VERSION 3.16)
project(paceval_hypothesis_testing LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# The official paceval_main.{h,cpp} is the API forwarding layer, not the
# mathematical runtime.  Link the compatible Linux .so separately.
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This tested reference configuration targets Linux")
endif()
if(NOT PACEVAL_RUNTIME AND DEFINED ENV{PACEVAL_LIB})
    set(PACEVAL_RUNTIME "$ENV{PACEVAL_LIB}")
endif()
if(NOT PACEVAL_RUNTIME OR NOT IS_ABSOLUTE "${PACEVAL_RUNTIME}" OR NOT EXISTS "${PACEVAL_RUNTIME}")
    message(FATAL_ERROR "Supply -DPACEVAL_RUNTIME=/absolute/path/libpaceval_linux_sharedLIB.so or PACEVAL_LIB")
endif()
add_library(paceval_external SHARED IMPORTED GLOBAL)
set_target_properties(paceval_external PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(hypothesis_test hypothesis_test.cpp paceval_main.cpp)
target_include_directories(hypothesis_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(hypothesis_test PRIVATE paceval_external)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(hypothesis_test PRIVATE -Wall -Wextra -Wpedantic)
    if(CMAKE_BUILD_TYPE STREQUAL "Release")
        target_compile_options(hypothesis_test PRIVATE -O3)
    endif()
endif()
include(CTest)
if(BUILD_TESTING)
    add_test(NAME analytic_reference_self_test COMMAND hypothesis_test --self-test)
    add_test(NAME real_paceval_pilot COMMAND hypothesis_test --candidates 256 --batch 37 --output "${CMAKE_CURRENT_BINARY_DIR}/smoke")
    add_test(NAME real_paceval_extended_pilot COMMAND hypothesis_test --candidates 512 --batch 128 --output "${CMAKE_CURRENT_BINARY_DIR}/extended")
endif()
