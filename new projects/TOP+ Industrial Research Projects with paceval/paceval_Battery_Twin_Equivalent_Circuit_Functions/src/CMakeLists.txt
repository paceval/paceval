cmake_minimum_required(VERSION 3.16)
project(paceval_battery_twin LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PACEVAL_RUNTIME "" CACHE FILEPATH "Path to externally supplied original paceval Linux shared library")
if(NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Provide -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(battery_twin battery_twin.cpp paceval_main.cpp)
target_compile_options(battery_twin PRIVATE -O2 -Wall -Wextra -Wpedantic)
target_link_libraries(battery_twin PRIVATE "${PACEVAL_RUNTIME}" dl pthread)
get_filename_component(PACEVAL_LIB_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
set_target_properties(battery_twin PROPERTIES BUILD_RPATH "${PACEVAL_LIB_DIR}")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
enable_testing()
add_test(NAME original_paceval_sdk_execution COMMAND battery_twin --out "${CMAKE_BINARY_DIR}/test_run" --steps 60)
set_tests_properties(original_paceval_sdk_execution PROPERTIES FIXTURES_SETUP test_data)
add_test(NAME independent_numerical_audit COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_results.py" --root "${CMAKE_BINARY_DIR}/test_run")
set_tests_properties(independent_numerical_audit PROPERTIES FIXTURES_REQUIRED test_data)
add_test(NAME github_readme_math_and_links COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_readme_math.py")
