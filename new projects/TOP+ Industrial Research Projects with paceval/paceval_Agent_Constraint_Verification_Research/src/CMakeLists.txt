cmake_minimum_required(VERSION 3.16)
project(paceval_agent_constraint_verification LANGUAGES CXX)
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Specify -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(agent_constraints agent_constraints.cpp paceval_main.cpp)
target_compile_features(agent_constraints PRIVATE cxx_std_17)
target_include_directories(agent_constraints PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(agent_constraints PRIVATE "${PACEVAL_RUNTIME}")
if(MSVC)
 target_compile_options(agent_constraints PRIVATE /W4)
else()
 target_compile_options(agent_constraints PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(UNIX AND NOT APPLE)
 get_filename_component(RUNTIME_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
 set_target_properties(agent_constraints PROPERTIES BUILD_RPATH "${RUNTIME_DIR}")
endif()
enable_testing()
add_test(NAME sdk_constraint_integration COMMAND agent_constraints --test --out "${CMAKE_CURRENT_BINARY_DIR}/test_outputs")
