cmake_minimum_required(VERSION 3.16)
project(paceval_robot_swept_volume LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Specify -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_external SHARED IMPORTED)
set_target_properties(paceval_external PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(robot_continuous robot_continuous_main.cpp paceval_main.cpp)
target_include_directories(robot_continuous PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(robot_continuous PRIVATE paceval_external)
if(UNIX AND NOT APPLE)
 target_link_libraries(robot_continuous PRIVATE dl pthread)
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
enable_testing()
add_test(NAME sdk_scalar_batch_operators_conversion COMMAND robot_continuous --selftest)
add_test(NAME full_csv_to_paceval_to_csv COMMAND robot_continuous --run "${CMAKE_CURRENT_LIST_DIR}/../input/cases.csv" "${CMAKE_CURRENT_BINARY_DIR}/ctest_results")
add_test(NAME independent_exact_interval_certificate COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/verify_interval_exact.py" "${CMAKE_CURRENT_BINARY_DIR}/ctest_results" "${CMAKE_CURRENT_LIST_DIR}/../input/cases.csv")
set_tests_properties(independent_exact_interval_certificate PROPERTIES DEPENDS full_csv_to_paceval_to_csv)
add_test(NAME independent_high_precision COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/verify_high_precision.py" "${CMAKE_CURRENT_BINARY_DIR}/ctest_results" "${CMAKE_CURRENT_LIST_DIR}/../input/cases.csv")
set_tests_properties(independent_high_precision PROPERTIES DEPENDS full_csv_to_paceval_to_csv)
add_test(NAME archive_markdown_math_and_links COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/check_markdown.py" "${CMAKE_CURRENT_LIST_DIR}/..")
add_test(NAME markdown_negative_regression COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/check_markdown.py" --selftest)
