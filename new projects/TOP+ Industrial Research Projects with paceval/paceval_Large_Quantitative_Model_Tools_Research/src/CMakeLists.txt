cmake_minimum_required(VERSION 3.16)
project(paceval_large_quantitative_tools LANGUAGES CXX)
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Provide -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(quant_tools quantitative_tools.cpp paceval_main.cpp)
target_compile_features(quant_tools PRIVATE cxx_std_17)
target_include_directories(quant_tools PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(quant_tools PRIVATE "${PACEVAL_RUNTIME}")
if(UNIX AND NOT APPLE)
 get_filename_component(RUNTIME_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
 set_target_properties(quant_tools PROPERTIES BUILD_RPATH "${RUNTIME_DIR}")
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(quant_tools PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME sdk_quantitative_tools COMMAND quant_tools --test --out "${CMAKE_CURRENT_BINARY_DIR}/test_outputs")
