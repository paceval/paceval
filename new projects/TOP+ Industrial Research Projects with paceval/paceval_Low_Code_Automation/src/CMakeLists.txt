cmake_minimum_required(VERSION 3.16)
project(paceval_low_code_automation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# The two official API files are compiled in this target. Their forwarding
# symbols are provided by the external, separately installed paceval engine.
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This reproducible reference configuration targets Linux")
endif()
if(NOT PACEVAL_RUNTIME AND DEFINED ENV{PACEVAL_LIB})
  set(PACEVAL_RUNTIME "$ENV{PACEVAL_LIB}")
endif()
if(NOT PACEVAL_RUNTIME OR NOT IS_ABSOLUTE "${PACEVAL_RUNTIME}" OR NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Supply -DPACEVAL_RUNTIME=/absolute/path/libpaceval_linux_sharedLIB.so or PACEVAL_LIB")
endif()
add_library(paceval_external SHARED IMPORTED GLOBAL)
set_target_properties(paceval_external PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(lowcode_math lowcode_math.cpp paceval_main.cpp)
target_include_directories(lowcode_math PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(lowcode_math PRIVATE paceval_external)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(lowcode_math PRIVATE -Wall -Wextra -Wpedantic)
  if(CMAKE_BUILD_TYPE STREQUAL "Release")
    target_compile_options(lowcode_math PRIVATE -O3)
  endif()
endif()
enable_testing()
add_test(NAME native_math_and_validation COMMAND lowcode_math --self-test)
add_test(NAME real_paceval_integration COMMAND lowcode_math --demo "${CMAKE_CURRENT_BINARY_DIR}/ctest_results.csv")

add_test(NAME real_paceval_tinc_audit_samples COMMAND lowcode_math --tinc "${CMAKE_CURRENT_BINARY_DIR}/ctest_tinc.csv")
