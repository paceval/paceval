cmake_minimum_required(VERSION 3.18)
project(industrial_piecewise_calibration_paceval LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PACEVAL_RUNTIME "" CACHE FILEPATH "Absolute path to the matching paceval Linux shared library")
if(NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Provide -DPACEVAL_RUNTIME=/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(industrial_calibration industrial_calibration.cpp paceval_main.cpp)
target_compile_options(industrial_calibration PRIVATE -O2 -Wall -Wextra -pedantic)
target_link_libraries(industrial_calibration PRIVATE "${PACEVAL_RUNTIME}" dl pthread)
get_filename_component(PACEVAL_LIB_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
set_target_properties(industrial_calibration PROPERTIES BUILD_RPATH "${PACEVAL_LIB_DIR}")
enable_testing()
add_test(NAME real_paceval_batch_128 COMMAND industrial_calibration ${CMAKE_SOURCE_DIR}/.. 128)
add_test(NAME real_paceval_batch_256 COMMAND industrial_calibration ${CMAKE_SOURCE_DIR}/.. 256)
add_test(NAME invalid_batch_rejected COMMAND industrial_calibration ${CMAKE_SOURCE_DIR}/.. 0)
set_tests_properties(invalid_batch_rejected PROPERTIES WILL_FAIL TRUE)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
 add_test(NAME independent_high_precision_audit COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/independent_reference.py ${CMAKE_SOURCE_DIR}/..)
endif()
if(Python3_Interpreter_FOUND)
 add_test(NAME graph_and_decision_inventory COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/verify_inventory.py ${CMAKE_SOURCE_DIR}/..)
endif()
if(Python3_Interpreter_FOUND)
 add_test(NAME version_manifest_hashes COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/verify_model_manifest.py ${CMAKE_SOURCE_DIR}/..)
endif()
if(Python3_Interpreter_FOUND)
 add_test(NAME rejects_invalid_domain_and_schema COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/invalid_input_tests.py $<TARGET_FILE:industrial_calibration>)
endif()
