cmake_minimum_required(VERSION 3.16)
project(paceval_sis_proof_test_reliability LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Set -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_runtime SHARED IMPORTED GLOBAL)
set_target_properties(paceval_runtime PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(sis_reliability sis_reliability.cpp paceval_main.cpp)
target_link_libraries(sis_reliability PRIVATE paceval_runtime dl)
set_target_properties(sis_reliability PROPERTIES BUILD_RPATH "$<TARGET_FILE_DIR:paceval_runtime>")
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(sis_reliability PRIVATE -O2 -Wall -Wextra)
endif()
enable_testing()
add_test(NAME original_sdk_converter_logic_and_scalar_batch COMMAND sis_reliability --selftest)
add_test(NAME original_sdk_small_functional_run COMMAND sis_reliability --run 160 "${CMAKE_BINARY_DIR}/small_results")
add_test(NAME original_sdk_medium_functional_run COMMAND sis_reliability --run 1200 "${CMAKE_BINARY_DIR}/medium_results")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME independent_high_precision_reference COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../validation/check_mpmath.py" "${CMAKE_CURRENT_SOURCE_DIR}/../results")
add_test(NAME archive_markdown_mathematics_links COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_archive_markdown.py")
add_test(NAME publication_pdf_and_safety_scope COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_publication.py")
add_test(NAME sha256_file_integrity COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_manifest.py")
