cmake_minimum_required(VERSION 3.16)
project(paceval_auction_bid_formulas LANGUAGES CXX)
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Specify -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(auction_bids auction_bids.cpp paceval_main.cpp)
target_compile_features(auction_bids PRIVATE cxx_std_17)
target_include_directories(auction_bids PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(auction_bids PRIVATE "${PACEVAL_RUNTIME}")
if(NOT MSVC)
 target_compile_options(auction_bids PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(UNIX AND NOT APPLE)
 get_filename_component(RUNTIME_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
 set_target_properties(auction_bids PROPERTIES BUILD_RPATH "${RUNTIME_DIR}")
endif()
enable_testing()
add_test(NAME sdk_auction_integration COMMAND auction_bids --out "${CMAKE_CURRENT_BINARY_DIR}/test_outputs")
