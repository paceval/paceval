cmake_minimum_required(VERSION 3.16)
project(paceval_reactor_screening LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Provide -DPACEVAL_RUNTIME=/absolute/path/to/compatible/libpaceval*.so")
endif()
add_library(paceval_runtime SHARED IMPORTED)
set_target_properties(paceval_runtime PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(reactor_screening reactor_main.cpp paceval_main.cpp)
target_include_directories(reactor_screening PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(reactor_screening PRIVATE paceval_runtime)
if(UNIX AND NOT APPLE)
  target_link_libraries(reactor_screening PRIVATE dl pthread)
endif()
enable_testing()
add_test(NAME sdk_operator_conversion_and_reference_tests COMMAND reactor_screening --selftest)
add_test(NAME readme_math_and_links COMMAND python3 "${CMAKE_CURRENT_LIST_DIR}/check_readme_math.py" "${CMAKE_CURRENT_LIST_DIR}/../README.md")
add_test(NAME readme_checker_regressions COMMAND python3 "${CMAKE_CURRENT_LIST_DIR}/check_readme_math.py" --selftest)
add_test(NAME full_csv_to_paceval_to_csv_pipeline
  COMMAND reactor_screening --run "${CMAKE_CURRENT_LIST_DIR}/../input/scenarios.csv" "${CMAKE_CURRENT_BINARY_DIR}/integration_results")
add_test(NAME independent_high_precision_research_audit
  COMMAND python3 "${CMAKE_CURRENT_LIST_DIR}/../scripts/verify_high_precision.py")
set_tests_properties(independent_high_precision_research_audit PROPERTIES DEPENDS full_csv_to_paceval_to_csv_pipeline)
