cmake_minimum_required(VERSION 3.16)
project(paceval_conservation_law_candidates LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PACEVAL_RUNTIME "" CACHE FILEPATH "Original paceval Linux dynamic library")
if(NOT EXISTS "${PACEVAL_RUNTIME}")
    message(FATAL_ERROR "Set -DPACEVAL_RUNTIME=/absolute/path/to/original/paceval.so")
endif()
add_executable(conservation_law_candidates conservation_law_candidates.cpp paceval_main.cpp)
target_compile_options(conservation_law_candidates PRIVATE -Wall -Wextra -Wpedantic -O2)
target_link_libraries(conservation_law_candidates PRIVATE "${PACEVAL_RUNTIME}" dl pthread)
get_filename_component(PACEVAL_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
set_target_properties(conservation_law_candidates PROPERTIES BUILD_RPATH "${PACEVAL_DIR}")
enable_testing()
add_test(NAME original_paceval_smoke COMMAND conservation_law_candidates --out "${CMAKE_CURRENT_BINARY_DIR}/smoke" --smoke)
