cmake_minimum_required(VERSION 3.18)
project(paceval_heat_exchanger_fouling LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PACEVAL_RUNTIME "" CACHE FILEPATH "Original ABI-compatible paceval Linux shared library, not included")
if(NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Provide original runtime -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(heat_exchanger_paceval heat_exchanger_paceval.cpp paceval_main.cpp)
target_link_libraries(heat_exchanger_paceval PRIVATE "${PACEVAL_RUNTIME}" dl pthread)
target_compile_options(heat_exchanger_paceval PRIVATE -Wall -Wextra -Wpedantic)
get_filename_component(PACEVAL_LIB_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
set_target_properties(heat_exchanger_paceval PROPERTIES BUILD_RPATH "${PACEVAL_LIB_DIR}")
enable_testing()
add_test(NAME original_sdk_conversions_and_operator_semantics COMMAND heat_exchanger_paceval "${CMAKE_SOURCE_DIR}/.." --selftest)
add_test(NAME original_sdk_complete_heat_exchanger_batch COMMAND heat_exchanger_paceval "${CMAKE_SOURCE_DIR}/..")
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME independently_recheck_archived_mcp_results COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/../validation/high_precision_reference.py" "${CMAKE_SOURCE_DIR}/..")
  add_test(NAME check_project_artifacts COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/../validation/check_project.py" "${CMAKE_SOURCE_DIR}/..")
endif()
