cmake_minimum_required(VERSION 3.16)
project(paceval_electric_motor_maps LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Provide -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_runtime SHARED IMPORTED)
set_target_properties(paceval_runtime PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(motor_maps motor_main.cpp paceval_main.cpp)
target_include_directories(motor_maps PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(motor_maps PRIVATE paceval_runtime)
if(UNIX AND NOT APPLE)
  target_link_libraries(motor_maps PRIVATE dl pthread)
endif()
enable_testing()
add_test(NAME sdk_scalar_batch_and_operator_tests COMMAND motor_maps --selftest)
add_test(NAME full_csv_to_paceval_pipeline COMMAND motor_maps --run "${CMAKE_CURRENT_LIST_DIR}/../input/candidates.csv" "${CMAKE_CURRENT_LIST_DIR}/../input/motor_parameters.csv" "${CMAKE_CURRENT_BINARY_DIR}/integration_results")
add_test(NAME independent_90_digit_verification COMMAND python3 "${CMAKE_CURRENT_LIST_DIR}/../scripts/verify_high_precision.py" "${CMAKE_CURRENT_BINARY_DIR}/integration_results" "${CMAKE_CURRENT_LIST_DIR}/../input/motor_parameters.csv")
set_tests_properties(independent_90_digit_verification PROPERTIES DEPENDS full_csv_to_paceval_pipeline)
add_test(NAME readme_math_and_relative_links COMMAND python3 "${CMAKE_CURRENT_LIST_DIR}/check_readme_math.py" "${CMAKE_CURRENT_LIST_DIR}/../README.md")
add_test(NAME readme_checker_negative_tests COMMAND python3 "${CMAKE_CURRENT_LIST_DIR}/check_readme_math.py" --selftest)
