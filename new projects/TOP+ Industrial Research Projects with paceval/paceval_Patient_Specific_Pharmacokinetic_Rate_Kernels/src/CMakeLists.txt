cmake_minimum_required(VERSION 3.16)
project(paceval_patient_specific_pharmacokinetics LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PACEVAL_RUNTIME "" CACHE FILEPATH "External original Linux paceval shared library; never distributed")
if(NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "An original Linux paceval shared library is required: -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(pharmacokinetic_kernels pharmacokinetic_kernels.cpp paceval_main.cpp)
target_compile_options(pharmacokinetic_kernels PRIVATE -O2 -Wall -Wextra -Wpedantic)
target_link_libraries(pharmacokinetic_kernels PRIVATE "${PACEVAL_RUNTIME}" dl pthread)
get_filename_component(PACEVAL_LIB_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
set_target_properties(pharmacokinetic_kernels PROPERTIES BUILD_RPATH "${PACEVAL_LIB_DIR}")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
enable_testing()
add_test(NAME original_paceval_pk_execution COMMAND pharmacokinetic_kernels --out "${CMAKE_BINARY_DIR}/test_run" --steps 96 --patients 4)
set_tests_properties(original_paceval_pk_execution PROPERTIES FIXTURES_SETUP pk_data)
add_test(NAME pk_independent_high_precision COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_results.py" --root "${CMAKE_BINARY_DIR}/test_run")
set_tests_properties(pk_independent_high_precision PROPERTIES FIXTURES_REQUIRED pk_data)
add_test(NAME github_readme_integrity COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_readme_math.py")
