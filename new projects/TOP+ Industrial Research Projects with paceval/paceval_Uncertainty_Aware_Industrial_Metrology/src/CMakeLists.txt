cmake_minimum_required(VERSION 3.16)
project(paceval_wp09_uncertainty_metrology LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Supply -DPACEVAL_RUNTIME=/absolute/path/to/compatible/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_runtime SHARED IMPORTED GLOBAL)
set_target_properties(paceval_runtime PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(metrology metrology.cpp paceval_main.cpp)
target_include_directories(metrology PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(metrology PRIVATE paceval_runtime dl)
target_compile_options(metrology PRIVATE -O2 -Wall -Wextra -Wpedantic)
set_target_properties(metrology PROPERTIES BUILD_RPATH "$<TARGET_FILE_DIR:paceval_runtime>")
enable_testing()
add_test(NAME strict_numeric_csv_conversion COMMAND metrology --test-conversion)
add_test(NAME mixed_paceval_operator_semantics COMMAND metrology --test-operators)
add_test(NAME scalar_multigraph_layout COMMAND metrology --test-layout)
add_test(NAME covariance_psd_domain_rejection COMMAND metrology --test-invalid)
add_test(NAME executable_synthetic_sample COMMAND metrology --run --revisions 2 --cases 3 --samples 128 --out "${CMAKE_BINARY_DIR}/sample_wp09")
add_test(NAME independent_sample_audit COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/verify_reference.py" "${CMAKE_BINARY_DIR}/sample_wp09" --small)
set_tests_properties(independent_sample_audit PROPERTIES DEPENDS executable_synthetic_sample)
add_test(NAME readme_links_math_and_hero COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/check_readme.py")
add_test(NAME readme_negative_link_regression COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/check_readme.py" --self-test)
add_test(NAME archive_checksum_manifest COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/verify_manifest.py")

add_test(NAME independent_high_precision_full_audit COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/verify_reference.py" "${CMAKE_CURRENT_SOURCE_DIR}/../results")
add_test(NAME independent_gaussian_monte_carlo COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/monte_carlo_check.py" "${CMAKE_CURRENT_SOURCE_DIR}/../results/audit_samples.csv")
