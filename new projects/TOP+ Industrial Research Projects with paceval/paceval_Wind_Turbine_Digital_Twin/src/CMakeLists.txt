cmake_minimum_required(VERSION 3.16)
project(paceval_wind_turbine_twin LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PACEVAL_RUNTIME "" CACHE FILEPATH "Original paceval Linux shared library (external; not redistributed)")
if(NOT EXISTS "${PACEVAL_RUNTIME}")
    message(FATAL_ERROR "Set -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(wind_turbine_twin wind_turbine_twin.cpp paceval_main.cpp)
target_compile_options(wind_turbine_twin PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(wind_turbine_twin PRIVATE "${PACEVAL_RUNTIME}" dl pthread)
get_filename_component(PACEVAL_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
set_target_properties(wind_turbine_twin PROPERTIES BUILD_RPATH "${PACEVAL_DIR}")
enable_testing()
add_test(NAME real_paceval_sdk_integration COMMAND wind_turbine_twin --out "${CMAKE_CURRENT_BINARY_DIR}/ctest" --steps 60 --surface 9 --sweep 60)
