cmake_minimum_required(VERSION 3.16)
project(paceval_portfolio_stress_research LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Compile the official C/C++ SDK facade directly into this demonstration.
# The architecture-specific paceval runtime is intentionally not redistributed.
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This demonstrator is validated on Linux; a compatible runtime is required")
endif()
if(NOT PACEVAL_RUNTIME AND DEFINED ENV{PACEVAL_LIB})
  set(PACEVAL_RUNTIME "$ENV{PACEVAL_LIB}")
endif()
if(NOT PACEVAL_RUNTIME OR NOT IS_ABSOLUTE "${PACEVAL_RUNTIME}" OR NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Pass -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_runtime SHARED IMPORTED GLOBAL)
set_target_properties(paceval_runtime PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(portfolio_stress portfolio_stress.cpp paceval_main.cpp)
target_include_directories(portfolio_stress PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(portfolio_stress PRIVATE paceval_runtime)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(portfolio_stress PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME independent_native_selftest COMMAND portfolio_stress --self-test)
  add_test(NAME sdk_smoke_100_models COMMAND portfolio_stress --scenarios 96 --models 100 --batch 37 --out "${CMAKE_CURRENT_BINARY_DIR}/smoke")
  add_test(NAME sdk_versioning_and_risk COMMAND portfolio_stress --scenarios 128 --models 40 --batch 32 --out "${CMAKE_CURRENT_BINARY_DIR}/versioning")
  add_test(NAME sdk_imported_scenarios COMMAND portfolio_stress --models 8 --input "${CMAKE_CURRENT_SOURCE_DIR}/../examples/factor_shocks.csv" --batch 37 --out "${CMAKE_CURRENT_BINARY_DIR}/import")
endif()
