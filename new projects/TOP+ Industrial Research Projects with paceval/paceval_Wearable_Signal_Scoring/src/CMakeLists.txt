cmake_minimum_required(VERSION 3.16)
project(paceval_wearable_scoring LANGUAGES CXX)
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Specify -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(wearable_scoring wearable_scoring.cpp paceval_main.cpp)
target_compile_features(wearable_scoring PRIVATE cxx_std_17)
target_include_directories(wearable_scoring PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(wearable_scoring PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(wearable_scoring PRIVATE "${PACEVAL_RUNTIME}")
if(UNIX AND NOT APPLE)
 get_filename_component(PACEVAL_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
 set_target_properties(wearable_scoring PROPERTIES BUILD_RPATH "${PACEVAL_DIR}")
endif()
enable_testing()
add_test(NAME wearable_original_sdk COMMAND wearable_scoring --windows 300 --out "${CMAKE_CURRENT_BINARY_DIR}/test_results")
