cmake_minimum_required(VERSION 3.16)
project(paceval_wafer_overlay LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT DEFINED PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Provide -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_runtime SHARED IMPORTED GLOBAL)
set_target_properties(paceval_runtime PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(wafer_overlay wafer_overlay.cpp paceval_main.cpp)
target_include_directories(wafer_overlay PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(wafer_overlay PRIVATE paceval_runtime dl)
target_compile_options(wafer_overlay PRIVATE -Wall -Wextra -Wpedantic -O2)
target_compile_definitions(wafer_overlay PRIVATE PROJECT_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/..")
set_target_properties(wafer_overlay PROPERTIES BUILD_RPATH "$<TARGET_FILE_DIR:paceval_runtime>")
enable_testing()
add_test(NAME operator_and_conversion_semantics COMMAND wafer_overlay --test-operators)
add_test(NAME invalid_numeric_input_rejected COMMAND wafer_overlay --test-invalid)
add_test(NAME scalar_batch_and_multigraph COMMAND wafer_overlay --test-layout)
add_test(NAME small_end_to_end_simulation COMMAND wafer_overlay --run --lot-limit 2 --grid-limit 128 --out "${CMAKE_BINARY_DIR}/test_run")
add_test(NAME numerical_output_audit COMMAND ${CMAKE_COMMAND} -E env PYTHONUNBUFFERED=1 python3 "${CMAKE_CURRENT_SOURCE_DIR}/verify_results.py" --project "${CMAKE_CURRENT_SOURCE_DIR}/.." --results "${CMAKE_BINARY_DIR}/test_run" --small)
set_tests_properties(numerical_output_audit PROPERTIES DEPENDS small_end_to_end_simulation)
add_test(NAME readme_math_and_links COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/check_readme_math.py" "${CMAKE_CURRENT_SOURCE_DIR}/../README.md")

add_test(NAME readme_negative_positive_regressions COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/check_readme_math.py" --self-test)
add_test(NAME package_manifest_integrity COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/verify_manifest.py" "${CMAKE_CURRENT_SOURCE_DIR}/..")
