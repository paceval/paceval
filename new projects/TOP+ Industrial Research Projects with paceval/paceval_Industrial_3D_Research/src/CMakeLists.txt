cmake_minimum_required(VERSION 3.16)
project(paceval_industrial_3d LANGUAGES CXX)
# Original public SDK facade, with its include path adapted to this directory.
# An ABI-compatible Linux runtime is supplied externally, not shipped.
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Supply -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(industrial_3d industrial_3d.cpp paceval_main.cpp)
target_compile_features(industrial_3d PRIVATE cxx_std_17)
target_include_directories(industrial_3d PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(industrial_3d PRIVATE "${PACEVAL_RUNTIME}")
if(MSVC)
 target_compile_options(industrial_3d PRIVATE /W4)
else()
 target_compile_options(industrial_3d PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(UNIX AND NOT APPLE)
 get_filename_component(PACEVAL_LIB_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
 set_target_properties(industrial_3d PROPERTIES BUILD_RPATH "${PACEVAL_LIB_DIR}")
endif()
enable_testing()
add_test(NAME six_industrial_surfaces COMMAND industrial_3d --out "${CMAKE_CURRENT_BINARY_DIR}/test_outputs")
