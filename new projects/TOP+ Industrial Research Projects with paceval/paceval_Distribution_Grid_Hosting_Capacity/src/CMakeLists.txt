cmake_minimum_required(VERSION 3.16)
project(paceval_wp11_grid_hosting LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Provide -DPACEVAL_RUNTIME=/absolute/path/to/compatible/libpaceval_linux_sharedLIB.so")
endif()
add_library(paceval_runtime SHARED IMPORTED GLOBAL)
set_target_properties(paceval_runtime PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(grid_hosting grid_hosting.cpp paceval_main.cpp)
target_include_directories(grid_hosting PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(grid_hosting PRIVATE paceval_runtime dl)
target_compile_options(grid_hosting PRIVATE -O2 -Wall -Wextra -Wpedantic)
set_target_properties(grid_hosting PROPERTIES BUILD_RPATH "$<TARGET_FILE_DIR:paceval_runtime>")
enable_testing()
add_test(NAME numeric_csv_paceval_conversions COMMAND grid_hosting --test-conversion)
add_test(NAME logical_relational_and_modulo_semantics COMMAND grid_hosting --test-operators)
add_test(NAME graph_major_multibatch_layout COMMAND grid_hosting --test-layout)
add_test(NAME overinjection_network_constraint COMMAND grid_hosting --test-invalid)
add_test(NAME full_small_feeder_research COMMAND grid_hosting --run --steps 5 --out "${CMAKE_BINARY_DIR}/small_results")
add_test(NAME small_independent_complex_and_scipy_check COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/../validation/verify_reference.py" "${CMAKE_BINARY_DIR}/small_results" --small)
set_tests_properties(small_independent_complex_and_scipy_check PROPERTIES DEPENDS full_small_feeder_research)
add_test(NAME independent_full_1024_case_audit COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/../validation/verify_reference.py" "${CMAKE_CURRENT_SOURCE_DIR}/../results")
add_test(NAME readme_links_math_check COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/check_readme.py")
add_test(NAME readme_link_negative_regression COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/check_readme.py" --self-test)
add_test(NAME whitepaper_pdf_page_and_cover_check COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/check_paper.py")
add_test(NAME release_checksum_manifest COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/verify_manifest.py")
