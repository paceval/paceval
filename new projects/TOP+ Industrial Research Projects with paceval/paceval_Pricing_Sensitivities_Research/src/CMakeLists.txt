cmake_minimum_required(VERSION 3.16)
project(paceval_pricing_sensitivities LANGUAGES CXX)
# Use the original paceval API facade, not a replacement or dlopen shim.
if(NOT PACEVAL_RUNTIME OR NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Set -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
add_executable(pricing_sensitivities pricing_sensitivities.cpp paceval_main.cpp)
target_compile_features(pricing_sensitivities PRIVATE cxx_std_17)
target_include_directories(pricing_sensitivities PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(pricing_sensitivities PRIVATE "${PACEVAL_RUNTIME}")
if(MSVC)
 target_compile_options(pricing_sensitivities PRIVATE /W4)
else()
 target_compile_options(pricing_sensitivities PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(UNIX AND NOT APPLE)
 get_filename_component(PACEVAL_RUNTIME_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
 set_target_properties(pricing_sensitivities PROPERTIES BUILD_RPATH "${PACEVAL_RUNTIME_DIR}")
endif()
enable_testing()
add_test(NAME integration_paceval_options COMMAND pricing_sensitivities --out "${CMAKE_CURRENT_BINARY_DIR}/test_outputs")
