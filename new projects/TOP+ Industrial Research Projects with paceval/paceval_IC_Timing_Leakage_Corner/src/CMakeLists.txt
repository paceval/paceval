cmake_minimum_required(VERSION 3.18)
project(paceval_ic_timing_leakage_corner_sweeps LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PACEVAL_RUNTIME "" CACHE FILEPATH "Original paceval Linux x86-64 shared library, ABI-compatible with included headers")
if(NOT EXISTS "${PACEVAL_RUNTIME}")
 message(FATAL_ERROR "Original paceval shared library required: -DPACEVAL_RUNTIME=/absolute/path/libpaceval_linux_sharedLIB.so")
endif()
add_executable(ic_corner_paceval ic_corner_paceval.cpp paceval_main.cpp test_numeric_conversions.cpp test_operator_semantics.cpp)
target_compile_options(ic_corner_paceval PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(ic_corner_paceval PRIVATE "${PACEVAL_RUNTIME}" dl pthread)
get_filename_component(PACEVAL_RUNTIME_DIR "${PACEVAL_RUNTIME}" DIRECTORY)
set_target_properties(ic_corner_paceval PROPERTIES BUILD_RPATH "${PACEVAL_RUNTIME_DIR}")
enable_testing()
add_test(NAME sdk_converters_and_operators COMMAND ic_corner_paceval "${CMAKE_SOURCE_DIR}/.." --selftest)
add_test(NAME sdk_corner_grid_and_csv COMMAND ic_corner_paceval "${CMAKE_SOURCE_DIR}/..")
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
 add_test(NAME independent_precision_audit COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/../validation/validate_records.py" "${CMAKE_SOURCE_DIR}/..")
 add_test(NAME project_artifact_integrity COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/../validation/check_project.py" "${CMAKE_SOURCE_DIR}/..")
endif()
