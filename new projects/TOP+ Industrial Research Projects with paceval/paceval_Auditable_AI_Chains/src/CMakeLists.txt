cmake_minimum_required(VERSION 3.16)
project(paceval_auditable_ai_chains VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This demonstration requires a compatible Linux paceval runtime")
endif()
if(NOT PACEVAL_RUNTIME AND DEFINED ENV{PACEVAL_LIB})
  set(PACEVAL_RUNTIME "$ENV{PACEVAL_LIB}")
endif()
if(NOT PACEVAL_RUNTIME OR NOT IS_ABSOLUTE "${PACEVAL_RUNTIME}" OR NOT EXISTS "${PACEVAL_RUNTIME}")
  message(FATAL_ERROR "Pass -DPACEVAL_RUNTIME=/absolute/path/to/libpaceval_linux_sharedLIB.so")
endif()
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(paceval_runtime SHARED IMPORTED GLOBAL)
set_target_properties(paceval_runtime PROPERTIES IMPORTED_LOCATION "${PACEVAL_RUNTIME}")
add_executable(auditable_chains auditable_chains.cpp paceval_main.cpp)
target_include_directories(auditable_chains PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(auditable_chains PRIVATE paceval_runtime OpenSSL::Crypto)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(auditable_chains PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME native_schema_checks COMMAND auditable_chains --self-test)
  add_test(NAME paceval_chain_cases COMMAND auditable_chains --demo --batch 37 --out "${CMAKE_CURRENT_BINARY_DIR}/suite")
  add_test(NAME paceval_batch_single COMMAND auditable_chains --demo --batch 1 --out "${CMAKE_CURRENT_BINARY_DIR}/single")
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME independent_high_precision_and_audit
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_audit.py" "${CMAKE_CURRENT_BINARY_DIR}/suite")
  add_test(NAME jsonl_adapter_integration
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/agent_tool_adapter.py"
            --binary "$<TARGET_FILE:auditable_chains>" --self-test)
  set_tests_properties(paceval_chain_cases PROPERTIES FIXTURES_SETUP generated_outputs)
  set_tests_properties(independent_high_precision_and_audit PROPERTIES FIXTURES_REQUIRED generated_outputs)
endif()
