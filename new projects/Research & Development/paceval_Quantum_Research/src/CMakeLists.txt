cmake_minimum_required(VERSION 3.12)
project(paceval_quantum_research LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(quantum quantum.cpp)
if(UNIX AND NOT APPLE)
  target_link_libraries(quantum PRIVATE ${CMAKE_DL_LIBS})
endif()
if(MSVC)
  target_compile_options(quantum PRIVATE /W4 /O2)
else()
  target_compile_options(quantum PRIVATE -O3 -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME native_self_test COMMAND quantum --self-test)
if(NOT PACEVAL_RUNTIME AND DEFINED ENV{PACEVAL_LIB})
  set(PACEVAL_RUNTIME "$ENV{PACEVAL_LIB}")
endif()
if(PACEVAL_RUNTIME)
  if(NOT EXISTS "${PACEVAL_RUNTIME}" OR IS_DIRECTORY "${PACEVAL_RUNTIME}")
    message(FATAL_ERROR "PACEVAL_RUNTIME must identify a real shared library: ${PACEVAL_RUNTIME}")
  endif()
  if(NOT IS_ABSOLUTE "${PACEVAL_RUNTIME}")
    message(FATAL_ERROR "PACEVAL_RUNTIME must be an absolute path")
  endif()
  add_test(NAME real_paceval_integration COMMAND quantum --library "${PACEVAL_RUNTIME}" --require-paceval
    --events 120 --batch 32 --input-csv "${CMAKE_CURRENT_SOURCE_DIR}/../examples/ramsey_circuit_sample.csv"
    --csv "${CMAKE_CURRENT_BINARY_DIR}/quantum_test_output.csv")
  message(STATUS "Actual paceval integration test configured with external runtime: ${PACEVAL_RUNTIME}")
else()
  message(STATUS "No runtime supplied: native self-test registered; real integration test disabled")
endif()
